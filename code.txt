import Phaser from 'phaser';

export class InputManager {
    constructor(scene) {
        this.scene = scene;
        this.cursors = scene.input.keyboard.createCursorKeys();

        this.keys = scene.input.keyboard.addKeys({
            enter: Phaser.Input.Keyboard.KeyCodes.ENTER,
            esc: Phaser.Input.Keyboard.KeyCodes.ESC,
            x: Phaser.Input.Keyboard.KeyCodes.X,
            y: Phaser.Input.Keyboard.KeyCodes.Y,
            tab: Phaser.Input.Keyboard.KeyCodes.TAB,
            next: Phaser.Input.Keyboard.KeyCodes.E,
            prev: Phaser.Input.Keyboard.KeyCodes.Q
        });

        this.pad = null;
        this.prevPadState = {};

        if (this.scene.input.gamepad.total > 0) {
            this.pad = this.scene.input.gamepad.getPad(0);
        }

        this.scene.input.gamepad.on('connected', (pad) => {
            this.pad = pad;
        });
    }

    isButtonDown(action) {
        if (action === 'X' && this.keys.x.isDown) return true;
        if (action === 'Y' && this.keys.y.isDown) return true;
        if (this.pad) {
            if (action === 'X' && this.pad.X) return true;
            if (action === 'Y' && this.pad.Y) return true;
        }
        return false;
    }

    getAction(action) {
        const justDown = (key) => Phaser.Input.Keyboard.JustDown(key);

        if (action === 'NEXT_TARGET') return justDown(this.keys.next) || this.isButtonPressed('R1');
        if (action === 'PREV_TARGET') return justDown(this.keys.prev) || this.isButtonPressed('L1');
        if (action === 'UP') return justDown(this.cursors.up) || this.isButtonPressed('up');
        if (action === 'DOWN') return justDown(this.cursors.down) || this.isButtonPressed('down');
        if (action === 'SELECT') return justDown(this.keys.enter) || this.isButtonPressed('A');
        if (action === 'BACK') return justDown(this.keys.esc) || this.isButtonPressed('B');

        return false;
    }

    isButtonPressed(button) {
        if (!this.pad) return false;
        const isDown = this.pad[button];
        const wasDown = this.prevPadState[button] || false;
        this.prevPadState[button] = isDown;
        return isDown && !wasDown;
    }

    getMovement() {
        let x = 0, y = 0;
        if (this.cursors.left.isDown) x = -1;
        else if (this.cursors.right.isDown) x = 1;
        if (this.cursors.up.isDown) y = -1;
        else if (this.cursors.down.isDown) y = 1;

        if (this.pad) {
            const t = 0.15;
            if (Math.abs(this.pad.leftStick.x) > t) x = this.pad.leftStick.x;
            if (Math.abs(this.pad.leftStick.y) > t) y = this.pad.leftStick.y;
        }
        return { x, y };
    }
}

import Phaser from 'phaser';

export const AbilityTargetingSystem = (entities, inputManager) => {
    const player = entities.find(e => e.components.PlayerTag);
    if (!player || !player.components.AbilityState || !player.components.Rotation) return;

    const state = player.components.AbilityState;
    const pPos = player.components.Position;
    const pRot = player.components.Rotation.angle; // Current heading in radians

    // 1. Armed State Logic
    const holdX = inputManager.isButtonDown('X');
    const holdY = inputManager.isButtonDown('Y');
    const prevArmed = state.armed;

    if (holdX && holdY) state.armed = null;
    else if (holdX) state.armed = 'X';
    else if (holdY) state.armed = 'Y';
    else state.armed = null;

    if (prevArmed !== state.armed) {
        state.targetIndex = 0;
        state.currentTargetId = null;
    }

    if (!state.armed) return;

    // 2. Filter Configuration
    const config = {
        'X': { types: ['A', 'B'], range: 400 * GRID.UNIT },
        'Y': { types: ['C'], range: 1000 * GRID.UNIT }
    }[state.armed];

    // 3. The 90-Degree Cone Filter
    const FOV = Math.PI / 2; // 90 degrees total
    const HALF_FOV = FOV / 2; // 45 degrees in either direction

    const validTargets = entities.filter(e => {
        if (e === player || !e.components.EntityType) return false;

        const typeMatch = config.types.includes(e.components.EntityType.value);
        const dist = Phaser.Math.Distance.Between(pPos.x, pPos.y, e.components.Position.x, e.components.Position.y);

        if (!typeMatch || dist > config.range) return false;

        // --- CONE LOGIC ---
        // 1. Find the angle from player to target
        const angleToTarget = Phaser.Math.Angle.Between(pPos.x, pPos.y, e.components.Position.x, e.components.Position.y);

        // 2. Find the difference between player heading and target angle
        // wrapAngle ensures the difference is always between -PI and PI
        const angleDiff = Phaser.Math.Angle.Wrap(angleToTarget - pRot);

        // 3. Check if the target is within the 45-degree offset
        return Math.abs(angleDiff) <= HALF_FOV;
    }).sort((a, b) => {
        // Spiral Sort (Rings of 50px + Clockwise Angle)
        const d1 = Phaser.Math.Distance.Between(pPos.x, pPos.y, a.components.Position.x, a.components.Position.y);
        const d2 = Phaser.Math.Distance.Between(pPos.x, pPos.y, b.components.Position.x, b.components.Position.y);

        const r1 = Math.floor(d1 / 50);
        const r2 = Math.floor(d2 / 50);
        if (r1 !== r2) return r1 - r2;

        const a1 = Math.atan2(a.components.Position.y - pPos.y, a.components.Position.x - pPos.x);
        const a2 = Math.atan2(b.components.Position.y - pPos.y, b.components.Position.x - pPos.x);
        return a1 - a2;
    });

    state.potentialTargets = validTargets.map(t => t.id);
    const count = state.potentialTargets.length;

    // 4. Cycling Logic
    if (count > 0) {
        if (inputManager.getAction('NEXT_TARGET')) {
            state.targetIndex = (state.targetIndex + 1) % count;
        } else if (inputManager.getAction('PREV_TARGET')) {
            state.targetIndex = (state.targetIndex - 1 + count) % count;
        }
        state.currentTargetId = state.potentialTargets[state.targetIndex];
    } else {
        state.currentTargetId = null;
    }
};

export const AttachmentSystem = (entities, inputManager) => {
    const player = entities.find(e => e.components.PlayerTag);
    if (!player) return;

    const pPos = player.components.Position;
    const parentComp = player.components.Parent;
    const bPressed = inputManager.getAction('BACK');

    // 1. If currently ATTACHED and B is pressed: DETACH
    if (parentComp && bPressed) {
        delete player.components.Parent;
        return;
    }

    // 2. If currently UNATTACHED and B is pressed: ATTEMPT ATTACH
    if (!parentComp && bPressed) {
        const platforms = entities.filter(e => e.components.PlatformTag && e.components.Collider);

        for (const plat of platforms) {
            const platPos = plat.components.Position;
            const col = plat.components.Collider;
            let isOver = false;

            // Check if player center is within platform bounds
            if (col.type === 'circle') {
                const dist = Phaser.Math.Distance.Between(pPos.x, pPos.y, platPos.x, platPos.y);
                isOver = dist <= col.radius;
            } else {
                isOver = (pPos.x >= platPos.x - col.width / 2 && pPos.x <= platPos.x + col.width / 2 &&
                    pPos.y >= platPos.y - col.height / 2 && pPos.y <= platPos.y + col.height / 2);
            }

            if (isOver) {
                // LOCK: Store the current relative offset
                player.components.Parent = {
                    _type: 'Parent',
                    id: plat.id,
                    offset: { x: pPos.x - platPos.x, y: pPos.y - platPos.y }
                };
                break; // Only attach to one platform at a time
            }
        }
    }
};

export const CameraSystem = (entities, scene) => {
    const cam = scene.cameras.main;
    const player = entities.find(e => e.components.PlayerTag && e.components.Position);

    // 1. No Player? Look at 0,0
    if (!player) {
        cam.centerOn(0, 0); // Or pan smoothly: cam.pan(0, 0, 1000);
        return;
    }

    const pPos = player.components.Position;
    let targetX = pPos.x;
    let targetY = pPos.y;

    // 2. Check for Active Target (Midpoint Logic)
    const abilityState = player.components.AbilityState;
    if (abilityState && abilityState.currentTargetId) {
        const target = entities.find(e => e.id === abilityState.currentTargetId);

        if (target && target.components.Position) {
            const tPos = target.components.Position;
            // Calculate Midpoint
            targetX = (pPos.x + tPos.x) / 2;
            targetY = (pPos.y + tPos.y) / 2;
        }
    }

    // 3. Move Camera
    // Lerp 0.1 gives a "spring" snap. Set to 1.0 for instant 1-frame snap.
    const lerp = 0.1;

    // Current camera center
    const currentX = cam.scrollX + cam.width / 2;
    const currentY = cam.scrollY + cam.height / 2;

    // Interpolate
    const newX = currentX + (targetX - currentX) * lerp;
    const newY = currentY + (targetY - currentY) * lerp;

    cam.centerOn(newX, newY);
};

export const DebugLayerSystem = (entities, scene) => {

    if (!scene.debugText) {
        scene.debugText = scene.add.text(10, 10, '', {

    }

    const cam = scene.cameras.main;
    const uiScale = 1 / cam.zoom;

    scene.debugText.setScale(uiScale);
    scene.debugText.setPosition(10 * uiScale, 10 * uiScale);

    const player = entities.find(e => e.components.PlayerTag);
    let debugInfo = `SCENE: ${scene.scene.key}\n`;

    if (player) {
        const { Position: pos, Parent: parent } = player.components;
        debugInfo += `POS: ${Math.round(pos.x)}, ${Math.round(pos.y)}\n`;

        if (parent) {
            debugInfo += `STATUS: ATTACHED\n`;
            debugInfo += `PARENT: ${parent.id}\n`;
            debugInfo += `OFFSET: ${Math.round(parent.offset.x)}, ${Math.round(parent.offset.y)}\n`;
        } else {
            debugInfo += `STATUS: FREE\n`;
        }
    }

    scene.debugText.setText(debugInfo);
};

import { RenderFactory } from './RenderFactory';

export const LayeredRenderSystem = (entities, scene) => {
    // 1. Find Player's View Layers
    const player = entities.find(e => e.components.PlayerTag);
    const viewLayers = player && player.components.ViewLayers
        ? player.components.ViewLayers.layers
        : ['BASE'];

    entities.forEach(entity => {
        const { Position: pos, Rotation: rot, Renderable: renderable } = entity.components;

        if (!renderable) return;

        const definitions = renderable.definitions;
        const activeGraphics = renderable.activeVisuals; // Holds instances of BaseGraphic

        // 2. Iterate over all defined layers for this entity
        Object.keys(definitions).forEach(layerKey => {
            const config = definitions[layerKey];

            // Logic: Is this layer visible to the player?
            const isVisible = viewLayers.some(view => layerKey === view || layerKey.startsWith(view + '_'));

            if (isVisible) {
                // CASE A: Should be visible
                if (!activeGraphics[layerKey]) {
                    // Create Graphic Instance
                    activeGraphics[layerKey] = RenderFactory.createGraphic(scene, { ...config, layer: layerKey });
                }

                // Delegate Update to the Graphic Instance
                activeGraphics[layerKey].update(pos, rot, config);
                activeGraphics[layerKey].setVisible(true);

            } else {
                // CASE B: Should NOT be visible
                if (activeGraphics[layerKey]) {
                    activeGraphics[layerKey].setVisible(false);
                }
            }
        });

        // 3. Cleanup "Stale" Graphics
        Object.keys(activeGraphics).forEach(activeKey => {
            if (!definitions[activeKey]) {
                activeGraphics[activeKey].destroy(); // The Graphic class cleans up its Phaser object
                delete activeGraphics[activeKey];
            }
        });
    });
};

import { GRID } from '../../core/Constants';

export const MovementSystem = (entities, inputManager) => {
    if (!inputManager) return;

    const speed = GRID.SPEED * GRID.UNIT;
    const move = inputManager.getMovement();

    entities.forEach(entity => {
        const { Position: pos, Velocity: vel, Rotation: rot, Parent: parent, InputControlled: input } = entity.components;

        if (input && pos) {
            // 1. Heading Rotation (Standard Twin-stick style: Left Stick moves & rotates)
            // Note: For true Twin-stick, we'd read Right Stick here separately.
            if (rot && (move.x !== 0 || move.y !== 0)) {
                rot.angle = Math.atan2(move.y, move.x);
            }

            // 2. Attached Movement (Locked to Bounds)
            if (parent) {
                const plat = entities.find(e => e.id === parent.id);
                const col = plat?.components.Collider;

                if (col) {
                    // Calculate potential new offset
                    // Note: We use fixed step logic in the scene, so '1.0' represents one tick here usually,
                    // but since we are multiplying by speed, we assume the caller handles the time step 
                    // or we normalize speed. For now, we assume simple accumulation.
                    let nextX = parent.offset.x + (move.x * speed * 0.016);
                    let nextY = parent.offset.y + (move.y * speed * 0.016);

                    let hitEdge = false;

                    // Clamp to Circular Platform
                    if (col.type === 'circle') {
                        const dist = Math.sqrt(nextX * nextX + nextY * nextY);
                        if (dist > col.radius) {
                            // Normalize the vector and multiply by radius to snap to edge
                            const angle = Math.atan2(nextY, nextX);
                            nextX = Math.cos(angle) * col.radius;
                            nextY = Math.sin(angle) * col.radius;
                            hitEdge = true;
                        }
                    }
                    // Clamp to Rectangular Platform
                    else {
                        const halfW = col.width / 2;
                        const halfH = col.height / 2;

                        if (nextX < -halfW || nextX > halfW) hitEdge = true;
                        if (nextY < -halfH || nextY > halfH) hitEdge = true;

                        nextX = Math.max(-halfW, Math.min(halfW, nextX));
                        nextY = Math.max(-halfH, Math.min(halfH, nextY));
                    }

                    parent.offset.x = nextX;
                    parent.offset.y = nextY;

                    // --- CONSTRAINT REACTION ---
                    if (hitEdge && plat.components.Graphic) {
                        // Randomize color to indicate "Struggling against barrier"
                        plat.components.Graphic.color = Math.random() * 0xffffff;
                    }
                }

                // Zero out velocity so PhysicsSystem doesn't double-move
                if (vel) { vel.x = 0; vel.y = 0; }
            }
            // 3. Free Movement (Fallback if not parented)
            else if (vel) {
                vel.x = move.x * speed;
                vel.y = move.y * speed;
            }
        }
    });
};

export const PhysicsSystem = (entities, delta) => {
    // Delta is now expected to be fixed (e.g., 16.66ms)
    const dt = delta / 1000;

    entities.forEach(entity => {
        const { Position: pos, Velocity: vel, Parent: parent } = entity.components;
        if (!pos) return;

        if (parent) {
            const platform = entities.find(e => e.id === parent.id);
            // Safety: Ensure platform still exists
            if (platform && platform.components.Position) {
                // Hard sync: World Pos = Platform World Pos + Local Offset
                pos.x = platform.components.Position.x + parent.offset.x;
                pos.y = platform.components.Position.y + parent.offset.y;
            } else {
                // Fallback: If platform died, detach? 
                // For now, do nothing or user remains at last pos.
            }
        } else if (vel) {
            // Standard Euler Integration
            pos.x += vel.x * dt;
            pos.y += vel.y * dt;
        }
    });
};

export const RenderSystem = (entities, scene) => {
    if (!scene.gfx) {
        // High base depth for the ECS graphics layer
        scene.gfx = scene.add.graphics().setDepth(1000);
    }
    scene.gfx.clear();

    // 1. Sort entities so Player is always drawn last (on top)
    const sortedEntities = [...entities].sort((a, b) => {
        const aIsPlayer = a.components.PlayerTag ? 1 : 0;

import Phaser from 'phaser';
import { BaseGraphic } from './BaseGraphic';

export class ChaosEmitterGraphic extends BaseGraphic {
    create(config) {
        this.gameObject = this.scene.add.circle(0, 0, (config.size || 1) * 10, config.coreColor || 0x00ffff);
        this.gameObject.setDepth(100);

        this.tracerGfx = this.scene.add.graphics().setDepth(90);
        this.particles = [];
        this.spawnTimer = 0;
        return this.gameObject;
    }

    onUpdate(config) {
        if (!this.gameObject) return;

        // Dynamic Spawning: higher intensity can spawn more/faster
        const intensity = config.intensity ?? 1.0;
        this.spawnTimer++;

        const spawnRate = config.spawnRate ?? (10 / intensity);
        const maxParticles = config.maxParticles ?? (20 * intensity);

        if (this.spawnTimer > spawnRate && this.particles.length < maxParticles) {
            this.spawnParticle(config);
            this.spawnTimer = 0;
        }

        this.tracerGfx.clear();

        for (let i = this.particles.length - 1; i >= 0; i--) {
            const p = this.particles[i];

            // Jitter and Push scaled by the live intensity value
            p.vx += (Math.random() - 0.5) * (config.jitter ?? 0.8) * intensity;
            p.vy += (Math.random() - 0.5) * (config.jitter ?? 0.8) * intensity;

            const dx = p.x - this.gameObject.x;
            const dy = p.y - this.gameObject.y;
            const dist = Math.sqrt(dx * dx + dy * dy);

            if (dist > 0) {
                const push = (config.push ?? 0.15) * intensity;
                p.vx += (dx / dist) * push;
                p.vy += (dy / dist) * push;
            }

            p.vx *= (config.friction ?? 0.92);
            p.vy *= (config.friction ?? 0.92);

            p.x += p.vx;
            p.y += p.vy;
            p.life -= 1;

            p.history.push({ x: p.x, y: p.y });
            if (p.history.length > (config.tailLen ?? 15)) p.history.shift();

            this.drawParticle(p, config, intensity);

            if (p.life <= 0) this.particles.splice(i, 1);
        }
    }

    drawParticle(p, config, intensity) {
        const alpha = (p.life / p.maxLife) * (config.alphaMult ?? 1.0);

        if (p.history.length > 1) {
            this.tracerGfx.lineStyle(config.thickness ?? 1, config.tracerColor ?? 0xffa500, alpha);
            this.tracerGfx.beginPath();
            this.tracerGfx.moveTo(p.history[0].x, p.history[0].y);
            p.history.forEach(pos => this.tracerGfx.lineTo(pos.x, pos.y));
            this.tracerGfx.strokePath();
        }

        this.tracerGfx.fillStyle(config.headColor ?? 0x00ffff, alpha);
        this.tracerGfx.fillCircle(p.x, p.y, config.headSize ?? 2);
    }

    spawnParticle(config) {
        const life = config.minLife ?? 40 + Math.random() * (config.lifeVar ?? 30);
        this.particles.push({
            x: this.gameObject.x,
            y: this.gameObject.y,
            vx: (Math.random() - 0.5) * (config.initialV ?? 2),
            vy: (Math.random() - 0.5) * (config.initialV ?? 2),
            history: [],
            life: life,
            maxLife: life
        });
    }

    destroy() {
        if (this.tracerGfx) this.tracerGfx.destroy();
        this.particles = [];
        super.destroy();
    }
}

            p.vy += Math.sin(time + noise) * (0.8 * intensity);

            const dx = p.x - x;
            const dy = p.y - y;
            const dist = Math.sqrt(dx * dx + dy * dy);

            if (dist > 0) {
                // Outward push scaled by intensity
                p.vx += (dx / dist) * (0.2 * intensity);
                p.vy += (dy / dist) * (0.2 * intensity);
            }

export class MenuScene extends Phaser.Scene {
    constructor() {
        super('MenuScene');
        this.options = ['START GAME', 'LOAD GAME', 'TEST SUITE'];
        this.selected = 0;
        this.textObjects = [];
    }

    create() {
        this.inputManager = new InputManager(this);
        this.textObjects = []; // CRITICAL: Reset the array

        this.add.text(100, 100, 'REFRACT', {
            fontSize: '64px',
            fill: '#fff',
            fontFamily: 'monospace'
        });

        this.options.forEach((opt, index) => {
            const text = this.add.text(100, 300 + (index * 60), `  ${opt}`, {
                fontSize: '32px',
                fill: '#666',
                fontFamily: 'monospace'
            });
            this.textObjects.push(text);
        });

        this.updateSelection();
    }

    update() {
        if (this.inputManager.getAction('UP')) {
            this.selected = (this.selected - 1 + this.options.length) % this.options.length;
            this.updateSelection();
        }
        else if (this.inputManager.getAction('DOWN')) {
            this.selected = (this.selected + 1) % this.options.length;
            this.updateSelection();
        }

        if (this.inputManager.getAction('SELECT')) {
            this.handleSelection();
        }
    }

    updateSelection() {
        this.textObjects.forEach((text, index) => {
            // Safety check to prevent the drawImage error during scene transitions
            if (!text || !text.active || !text.canvas) return;

            if (index === this.selected) {
                text.setColor('#0f0');
                text.setText(`> ${this.options[index]}`);
            } else {
                text.setColor('#666');
                text.setText(`  ${this.options[index]}`);
            }
        });
    }

    handleSelection() {
        const choice = this.options[this.selected];
        if (choice === 'TEST SUITE') {
            this.scene.start('TestMenuScene');
        } else {
            console.log("Not implemented yet:", choice);
        }
    }
}

                }

    }

        const player = this.entities.find(e => e.components.PlayerTag);
        if (!player) return;

        const pPos = player.components.Position;
        const activationRange = 250; // Distance where effect starts growing

        // Update all reactive emitters
        this.entities.forEach(entity => {
            const renderable = entity.components.Renderable;
            if (!renderable || entity.components.PlayerTag) return;

            const config = renderable.definitions['BASE'];
            if (config && (config.type === 'chaos_emitter' || config.type === 'fairy_glow')) {
                const ePos = entity.components.Position;
                const dist = Phaser.Math.Distance.Between(pPos.x, pPos.y, ePos.x, ePos.y);

                // Map distance to intensity: 
                // Close (0px) = 2.0 intensity
                // Far (250px+) = 0.2 intensity
                let intensity = Phaser.Math.Clamp(1 - (dist / activationRange), 0, 1);

                // Scale it to a nice range (0.2 is calm, 2.0 is frantic)
                config.intensity = 0.2 + (intensity * 1.8);
            }
        });

import {

export class BaseTestScene extends Phaser.Scene {
    constructor(key, title, subtitle) {
        super(key);
        this.testTitle = title || "Test Scene";
        this.testSubtitle = subtitle || "No description provided.";
        this.accumulator = 0;
    }

    create() {
        this.entities = [];
        this.inputManager = new InputManager(this);

        const titleStr = this.testTitle || 'SCENE';
        UI.showTitle(titleStr, this.testSubtitle);

        this.input.keyboard.on('keydown-TAB', () => {
            this.showDebugGrid = !this.showDebugGrid;
            UI.toggleDebug();
        });

        this.input.keyboard.on('keydown-ESC', () => {
            this.scene.start('TestMenuScene');
        });
    }

    update(time, delta) {
        this.accumulator += delta;

        while (this.accumulator >= PHYSICS.STEP) {
            const dtSeconds = PHYSICS.STEP / 1000;

            // Logic Systems
            MovementSystem(this.entities, this.inputManager, dtSeconds);
            AttachmentSystem(this.entities, this.inputManager);
            AbilityTargetingSystem(this.entities, this.inputManager);

            this.fixedUpdate(time, PHYSICS.STEP);
            PhysicsSystem(this.entities, PHYSICS.STEP);

            this.accumulator -= PHYSICS.STEP;
        }

        // --- RENDER LOOP ---
        TargetRenderSystem(this.entities, this);
        CameraSystem(this.entities, this);

        // CRITICAL FIX: Use LayeredRenderSystem instead of RenderSystem
        LayeredRenderSystem(this.entities, this);

        DebugLayerSystem(this.entities, this);
        this.updateDebugUI();
    }

    fixedUpdate(time, dt) { }

    updateDebugUI() {
        const player = this.entities.find(e => e.components.PlayerTag);
        if (player) {
            const p = player.components.Position;
            const t = player.components.AbilityState?.currentTargetId
                ? { id: player.components.AbilityState.currentTargetId }
                : null;

            UI.updateDebug({
                x: p.x.toFixed(0),
                y: p.y.toFixed(0),
                gx: (p.x / GRID.UNIT).toFixed(1),
                gy: (p.y / GRID.UNIT).toFixed(1)
            }, t);
        }
    }

    addEntity(componentsArray) {
        const config = componentsArray.find(c => c.id && !c._type);
        const entity = {
            id: config ? config.id : crypto.randomUUID(),
            components: {}
        };
        componentsArray.forEach(c => {
            if (c && c._type) entity.components[c._type] = c;
        });
        this.entities.push(entity);
        return entity;
    }
}

import Phaser from 'phaser';
import { InputManager } from '../core/Input';
import { TestScenes } from './test/index';

export class TestMenuScene extends Phaser.Scene {
    constructor() {
        super('TestMenuScene');
        this.selected = 0;
        this.textObjects = [];
    }

    create() {
        this.inputManager = new InputManager(this);
        this.textObjects = [];

        // Background / UI Header
        this.add.text(50, 50, 'LAB / TEST SUITE', {
            fontSize: '32px',
            fill: '#0ff',
            fontFamily: 'monospace'
        });

        // Generate menu items from the registry
        TestScenes.forEach((sceneClass, index) => {
            const name = sceneClass.name;
            const text = this.add.text(100, 150 + (index * 50), `  ${name}`, {
                fontSize: '24px',
                fill: '#666',
                fontFamily: 'monospace'
            });
            this.textObjects.push(text);
        });

        this.updateSelection();
    }

    update() {
        // Navigation
        if (this.inputManager.getAction('UP')) {
            this.selected = (this.selected - 1 + TestScenes.length) % TestScenes.length;
            this.updateSelection();
        }
        else if (this.inputManager.getAction('DOWN')) {
            this.selected = (this.selected + 1) % TestScenes.length;
            this.updateSelection();
        }

        // Action
        if (this.inputManager.getAction('SELECT')) {
            const sceneClass = TestScenes[this.selected];
            this.scene.start(sceneClass.name);
        }

        if (this.inputManager.getAction('BACK')) {
            this.scene.start('MenuScene');
        }
    }

    updateSelection() {
        this.textObjects.forEach((text, index) => {
            if (!text || !text.active) return;
            const sceneName = TestScenes[index].name;
            if (index === this.selected) {
                text.setColor('#0f0').setText(`> ${sceneName}`);
            } else {
                text.setColor('#666').setText(`  ${sceneName}`);
            }
        });
    }
}
