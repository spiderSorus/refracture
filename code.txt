module.exports = class SceneManager {
    constructor(app) {
        this.app = app;
        this.currentScene = null;
    }

    async loadScene(SceneClass) {
        console.log(`SceneManager: Loading ${SceneClass.name}...`);

        // 1. Cleanup old scene
        if (this.currentScene) {
            this.app.ticker.remove(this.currentScene.update, this.currentScene);
            this.currentScene.destroy();
        }

        // 2. Initialize new scene
        this.currentScene = new SceneClass(this.app, this);
        await this.currentScene.init();

        // 3. Re-attach ticker
        // We bind 'this' so the scene update has access to its own properties
        this.app.ticker.add(this.currentScene.update, this.currentScene);
    }
};

const { defineQuery } = require('bitecs');
const { Position, CameraTarget } = require('../components');
const { PPU } = require('../../constants');

const targetQuery = defineQuery([Position, CameraTarget]);

module.exports = function createCameraSystem(app) {
    return (world) => {
        const ents = targetQuery(world);
        if (ents.length === 0) return;

        let targetX, targetY;
        let zoom = 1.0;

        if (ents.length === 1) {
            targetX = Position.x[ents[0]] * PPU;
            targetY = Position.y[ents[0]] * PPU;
        } else {
            const e1 = ents[0];
            const e2 = ents[1];
            const p1x = Position.x[e1] * PPU;
            const p1y = Position.y[e1] * PPU;
            const p2x = Position.x[e2] * PPU;
            const p2y = Position.y[e2] * PPU;

            targetX = (p1x + p2x) / 2;
            targetY = (p1y + p2y) / 2;

            const dx = Math.abs(p1x - p2x);
            const dy = Math.abs(p1y - p2y);
            const padding = 200; 
            
            const zoomX = app.screen.width / (dx + padding);
            const zoomY = app.screen.height / (dy + padding);
            zoom = Math.min(zoomX, zoomY, 1.0);
        }

        const lerp = 0.1;
        
        // Smooth zoom
        app.stage.scale.set(
            app.stage.scale.x + (zoom - app.stage.scale.x) * lerp
        );
        
        // Smooth pivot (looking at target)
        app.stage.pivot.x += (targetX - app.stage.pivot.x) * lerp;
        app.stage.pivot.y += (targetY - app.stage.pivot.y) * lerp;
        
        // Offset stage to center the pivot on screen
        app.stage.position.x = app.screen.width / 2;
        app.stage.position.y = app.screen.height / 2;
    };
};

const { defineQuery } = require('bitecs');
const { Position, Velocity, Mass } = require('../components');

// We want anything that has Mass, Position, and Velocity
const gravityQuery = defineQuery([Position, Velocity, Mass]);

const G = 500; // Gravitational Constant (Tweaked for game feel)
const MIN_DIST = 20; // Softening parameter to prevent shooting off to infinity

module.exports = function createGravitySystem() {
    return (world, dt) => {
        const ents = gravityQuery(world);

        // N-Body Simulation: Check every pair of objects
        for (let i = 0; i < ents.length; i++) {
            const e1 = ents[i];

            for (let j = i + 1; j < ents.length; j++) {
                const e2 = ents[j];

                const dx = Position.x[e2] - Position.x[e1];
                const dy = Position.y[e2] - Position.y[e1];
                const distSq = dx * dx + dy * dy;
                const dist = Math.sqrt(distSq);

                // Prevent division by zero or extreme forces at close range
                if (dist < MIN_DIST) continue;

                // F = G * m1 * m2 / r^2
                const f = (G * Mass.value[e1] * Mass.value[e2]) / distSq;

                // Normalize direction (dx/dist, dy/dist)
                const fx = (dx / dist) * f;
                const fy = (dy / dist) * f;

                // Apply Force to Velocity (a = F/m) -> (dv = a * dt)
                // Entity 1 is pulled TOWARDS Entity 2 (+)
                Velocity.x[e1] += (fx / Mass.value[e1]) * dt;
                Velocity.y[e1] += (fy / Mass.value[e1]) * dt;

                // Entity 2 is pulled TOWARDS Entity 1 (-) (Newton's 3rd Law)
                Velocity.x[e2] -= (fx / Mass.value[e2]) * dt;
                Velocity.y[e2] -= (fy / Mass.value[e2]) * dt;
            }
        }
    };
};

const { defineQuery } = require('bitecs');
const { Velocity, Input } = require('../components');
const { PLAYER_SPEED } = require('../../constants');

const inputQuery = defineQuery([Input, Velocity]);

module.exports = function createInputSystem() {
    return (world) => {
        const ents = inputQuery(world);
        const gamepads = navigator.getGamepads();

        for (let i = 0; i < ents.length; i++) {
            const eid = ents[i];
            const gp = gamepads[Input.index[eid]];

            if (gp) {
                // Movement
                const threshold = 0.15;
                const lx = gp.axes[0];
                const ly = gp.axes[1];
                Velocity.x[eid] = Math.abs(lx) > threshold ? lx * PLAYER_SPEED : 0;
                Velocity.y[eid] = Math.abs(ly) > threshold ? ly * PLAYER_SPEED : 0;

                // Triggers (R2 is button 7 on standard gamepads)
                // If keyboard override is needed later, we can map Shift/Ctrl here too.
                const r2 = gp.buttons[7] ? gp.buttons[7].value : 0;
                Input.r2[eid] = r2;
            }
        }
    };
};

module.exports = function createKeyboardSystem() {
    const keyMap = {};

    window.addEventListener('keydown', (e) => {
        keyMap[e.code] = 1;
        // Prevent scrolling with arrows
        if (['ArrowUp', 'ArrowDown', 'Space', 'Enter'].includes(e.code)) e.preventDefault();
    });

    window.addEventListener('keyup', (e) => {
        keyMap[e.code] = 0;
    });

    return (world) => {
        const ents = keyboardQuery(world);
        for (let i = 0; i < ents.length; i++) {
            const eid = ents[i];

            // Map standard keys to array indices
            Keyboard.keys[eid][32] = keyMap['Space'] || 0;
            Keyboard.keys[eid][13] = keyMap['Enter'] || 0; // Enter
            Keyboard.keys[eid][38] = keyMap['ArrowUp'] || 0; // Up
            Keyboard.keys[eid][40] = keyMap['ArrowDown'] || 0; // Down
            Keyboard.keys[eid][88] = keyMap['KeyX'] || 0; // 'X' key
        }
    };
};

const { defineQuery, addComponent, removeComponent, hasComponent } = require('bitecs');
const { Position, Velocity, Platform, Input, Parent } = require('../components');

const platformQuery = defineQuery([Position, Velocity, Platform]);
const playerQuery = defineQuery([Position, Input]);

module.exports = function createPlatformSystem() {
    return (world, dt) => {
        const platforms = platformQuery(world);
        const players = playerQuery(world);

        // 1. Platform Logic (Simple Bounce)
        for (let i = 0; i < platforms.length; i++) {
            const pid = platforms[i];
            if (Position.x[pid] > 200 && Velocity.x[pid] > 0) Velocity.x[pid] *= -1;
            if (Position.x[pid] < 0 && Velocity.x[pid] < 0) Velocity.x[pid] *= -1;
        }

        // 2. Player Logic
        for (let i = 0; i < players.length; i++) {
            const eid = players[i];

            // --- PARENTED BEHAVIOR ---
            if (hasComponent(world, Parent, eid)) {
                const pid = Parent.eid[eid];

                // A. Apply Platform Velocity (Carry)
                const pVx = Velocity.x[pid];
                const pVy = Velocity.y[pid];
                Position.x[eid] += pVx * dt;
                Position.y[eid] += pVy * dt;

                // B. Calculate Relative Position (The "Anchor")
                // Currently, the anchor is just the player's center (Position)
                let localX = Position.x[eid] - Position.x[pid];
                let localY = Position.y[eid] - Position.y[pid];

                // C. Check Bounds
                const halfW = Platform.width[pid] / 2;
                const halfH = Platform.height[pid] / 2;

                // Check if R2 is held (Threshold 0.1)
                const isSafetyOn = Input.r2[eid] > 0.1;

                if (isSafetyOn) {
                    // --- SAFETY MODE: CLAMP ---
                    // If we are about to leave, force position to the edge
                    if (localX < -halfW) localX = -halfW;
                    if (localX > halfW) localX = halfW;
                    if (localY < -halfH) localY = -halfH;
                    if (localY > halfH) localY = halfH;

                    // Apply Clamped Position back to World
                    Position.x[eid] = Position.x[pid] + localX;
                    Position.y[eid] = Position.y[pid] + localY;

                    // (Optional) Kill Player Velocity into the wall so it feels "solid"
                    // Velocity.x[eid] = 0; 
                }
                else {
                    // --- NORMAL MODE: DETACH ---
                    // If outside bounds, let them fall off
                    if (Math.abs(localX) > halfW || Math.abs(localY) > halfH) {
                        console.log("Player detached!");
                        removeComponent(world, Parent, eid);
                    }
                }

            } else {
                // --- ATTACH BEHAVIOR ---
                for (let j = 0; j < platforms.length; j++) {
                    const pid = platforms[j];
                    const dx = Math.abs(Position.x[eid] - Position.x[pid]);
                    const dy = Math.abs(Position.y[eid] - Position.y[pid]);
                    const w = Platform.width[pid] / 2;
                    const h = Platform.height[pid] / 2;

                    if (dx < w && dy < h) {
                        console.log("Player attached!");
                        addComponent(world, Parent, eid);
                        Parent.eid[eid] = pid;
                        break;
                    }
                }
            }
        }
    };
};

const { defineQuery, enterQuery, exitQuery, hasComponent } = require('bitecs');
const { Position, Graphic, Input, Debug, Platform } = require('../components'); // Added Platform
const { PPU } = require('../../constants');
const PIXI = require('pixi.js');

const controllers = {
    2: require('../../graphics/platform')
};

const renderQuery = defineQuery([Position, Graphic]);
const renderEnterQuery = enterQuery(renderQuery);
const renderExitQuery = exitQuery(renderQuery);
const playerQuery = defineQuery([Input, Position]);

const sprites = new Map();

module.exports = function createRenderSystem(app, state) {
    const worldLayer = app.stage;

    // --- Z-INDEX FIX ---
    // Allow us to reorder sprites (Player on top of Platform)
    worldLayer.sortableChildren = true;

    const uiLayer = new PIXI.Container();
    app.stage.addChild(uiLayer);

    const crosshair = new PIXI.Graphics();
    worldLayer.addChildAt(crosshair, 0);

    const textStyle = new PIXI.TextStyle({
        fontFamily: 'monospace', fontSize: 20, fill: '#FFB000', fontWeight: 'bold',
        stroke: { color: '#000000', width: 4 },
    });
    const playerText = new PIXI.Text({ text: '', style: textStyle });
    uiLayer.addChild(playerText);

    function updateHUD(world) {
        if (!state.debug) {
            uiLayer.visible = false;
            return;
        }
        uiLayer.visible = true;
        if (app.stage.getChildIndex(uiLayer) !== app.stage.children.length - 1) {
            app.stage.setChildIndex(uiLayer, app.stage.children.length - 1);
        }

        const players = playerQuery(world);
        if (players.length > 0) {
            const p = players[0];
            playerText.text = `POS: ${Position.x[p].toFixed(0)}, ${Position.y[p].toFixed(0)}`;
            playerText.position.set(20, app.screen.height - 50);
        }

        crosshair.clear();
        const zoom = worldLayer.scale.x;
        const size = 30 / zoom;
        crosshair.moveTo(0, -size).lineTo(0, size).stroke({ width: 2 / zoom, color: 0x00FF00 });
        crosshair.moveTo(-size, 0).lineTo(size, 0).stroke({ width: 2 / zoom, color: 0x00FF00 });
    }

    return (world) => {
        const dt = app.ticker.deltaMS / 1000;
        updateHUD(world);

        // 1. ENTER
        const entered = renderEnterQuery(world);
        for (let i = 0; i < entered.length; i++) {
            const eid = entered[i];
            const g = new PIXI.Graphics();

            // --- Z-INDEX ASSIGNMENT ---
            // Platforms = 0 (Background)
            // Players/Others = 10 (Foreground)
            const isPlatform = hasComponent(world, Platform, eid);
            g.zIndex = isPlatform ? 0 : 10;

            worldLayer.addChild(g);
            sprites.set(eid, g);
            Graphic.lastFrame[eid] = 65535;
        }

        // 2. UPDATE
        const ents = renderQuery(world);
        for (let i = 0; i < ents.length; i++) {
            const eid = ents[i];
            const g = sprites.get(eid);
            if (!g) continue;

            // Debug visibility
            g.visible = hasComponent(world, Debug, eid) ? state.debug : true;
            if (!g.visible) continue;

            g.x = Position.x[eid] * PPU;
            g.y = Position.y[eid] * PPU;

            // Controller Logic
            const cId = Graphic.controllerId[eid];
            const ctrl = controllers[cId];

            if (ctrl) {
                const currentFrame = Graphic.frame[eid];
                const lastFrame = Graphic.lastFrame[eid];

                if (currentFrame !== lastFrame) {
                    if (ctrl.onFrameChange) ctrl.onFrameChange(eid, lastFrame, currentFrame);
                    Graphic.lastFrame[eid] = currentFrame;
                }

                ctrl.update(g, eid, currentFrame, PPU, dt);
            } else {
                // Fallback
                g.clear();
                const size = Graphic.size[eid] * PPU;
                if (Graphic.shape[eid] === 0) g.circle(0, 0, size / 2).fill(Graphic.color[eid]);
                else g.rect(-size / 2, -size / 2, size, size).fill(Graphic.color[eid]);
            }
        }

        // 3. EXIT
        const exited = renderExitQuery(world);
        for (let i = 0; i < exited.length; i++) {
            const eid = exited[i];
            const g = sprites.get(eid);
            if (g) {
                worldLayer.removeChild(g);
                g.destroy();
                sprites.delete(eid);
            }
        }
    };
};

const PIXI = require('pixi.js');
const MenuScene = require('./MenuScene');

module.exports = class LoadingScene {
    constructor(app, manager) {
        this.app = app;
        this.manager = manager;
        this.box = new PIXI.Graphics();
        this.currentSize = 0;
        this.targetSize = 200;
        this.growthSpeed = 300; // pixels per second
    }

    async init() {
        console.log("LoadingScene: Initializing Centered Growth...");

        // Draw a 1x1 rect so we can scale it effectively, 
        // or just redraw the rect in the update loop.
        // We will position it in the center of the screen.
        this.box.x = this.app.screen.width / 2;
        this.box.y = this.app.screen.height / 2;

        this.app.stage.addChild(this.box);
    }

    update() {
        const dt = this.app.ticker.deltaMS / 1000;

        if (this.currentSize < this.targetSize) {
            this.currentSize += this.growthSpeed * dt;

            // Re-draw the box centered on its own coordinates
            this.box.clear();
            this.box.rect(
                -this.currentSize / 2,
                -this.currentSize / 2,
                this.currentSize,
                this.currentSize
            ).fill(0x0000FF);
        } else {
            // Once we hit 200px, transition
            this.manager.loadScene(MenuScene);
        }
    }

    destroy() {
        console.log("LoadingScene: Transitioning to Menu...");
        this.app.stage.removeChild(this.box);
        this.box.destroy();
    }
};

        this.selectionIndex = 0;

        // Input Edge Detection
        this.lastInput = { up: 0, down: 0, confirm: 0 };

        // Data for Menus
        this.mainMenuItems = [
            { label: "< NEW >", action: () => console.log("New Game - TODO") },
            { label: "< LOAD >", action: () => console.log("Load Game - TODO") },
            { label: "< TEST >", action: () => this.switchState(STATE_TEST) }
        ];
    }

    async init() {
        // Setup Input Entity
        const eid = addEntity(this.world);
        addComponent(this.world, Keyboard, eid);

        // Setup Visual Containers
        this.container = new PIXI.Container();
        this.app.stage.addChild(this.container);

        this.titleText = new PIXI.Text({
            text: 'REFRACT', style: {
                fontFamily: 'monospace', fontSize: 60, fill: '#FFB000', fontWeight: 'bold'
            }
        });
        this.titleText.anchor.set(0.5);
        this.titleText.x = this.app.screen.width / 2;
        this.titleText.y = 100;
        this.container.addChild(this.titleText);

        this.menuText = new PIXI.Text({
            text: '', style: {
                fontFamily: 'monospace', fontSize: 30, fill: '#FFFFFF', align: 'center', lineHeight: 50
            }
        });
        this.menuText.anchor.set(0.5, 0);
        this.menuText.x = this.app.screen.width / 2;
        this.menuText.y = 250;
        this.container.addChild(this.menuText);

        this.renderMenu();
    }

    switchState(newState) {
        this.menuState = newState;
        this.selectionIndex = 0; // Reset selection on state change
        this.renderMenu();
    }

    // "Ring Buffer" Logic
    moveSelection(dir, length) {
        this.selectionIndex += dir;
        if (this.selectionIndex < 0) this.selectionIndex = length - 1;
        if (this.selectionIndex >= length) this.selectionIndex = 0;
        this.renderMenu();
    }

    update() {
        this.keyboardSystem(this.world);
        const ents = keyboardQuery(this.world);
        if (ents.length === 0) return;

        const eid = ents[0];
        const keys = Keyboard.keys[eid];

        // Input Mapping
        const up = keys[38];
        const down = keys[40];
        // Confirm is Space (32), Enter (13), or X (88)
        const confirm = keys[32] || keys[13] || keys[88];

        // 1. Handle Navigation (Up/Down) - Only on Edge (Press)
        if (up && !this.lastInput.up) {
            if (this.menuState === STATE_MAIN) this.moveSelection(-1, this.mainMenuItems.length);
            if (this.menuState === STATE_TEST) this.moveSelection(-1, testScenes.length);
        }
        if (down && !this.lastInput.down) {
            if (this.menuState === STATE_MAIN) this.moveSelection(1, this.mainMenuItems.length);
            if (this.menuState === STATE_TEST) this.moveSelection(1, testScenes.length);
        }

        // 2. Handle Confirmation
        if (confirm && !this.lastInput.confirm) {
            if (this.menuState === STATE_WAITING) {
                this.switchState(STATE_MAIN);
            } else if (this.menuState === STATE_MAIN) {
                // Execute the action for the selected item
                this.mainMenuItems[this.selectionIndex].action();
            } else if (this.menuState === STATE_TEST) {
                // Load the selected test scene
                const selected = testScenes[this.selectionIndex];
                if (selected && selected.SceneClass) {
                    console.log(`Loading ${selected.label}...`);
                    try {
                        this.manager.loadScene(selected.SceneClass);
                    } catch (e) {
                        console.error("Failed to load scene:", e);
                    }
                }
            }
        }

        // Update Edge State
        this.lastInput = { up, down, confirm };
    }

    renderMenu() {
        if (this.menuState === STATE_WAITING) {
            this.menuText.text = "[ PRESS SPACE TO START ]";
            return;
        }

        let items = [];
        if (this.menuState === STATE_MAIN) items = this.mainMenuItems;
        else if (this.menuState === STATE_TEST) items = testScenes;

        // Generate Text String
        const textStr = items.map((item, index) => {
            let label = item.label;

            // AUTO-NUMBERING: If we are in the Test Menu, prepend the index
            if (this.menuState === STATE_TEST) {
                const num = String(index).padStart(2, '0'); // "00", "01", etc.
                label = `${num} ${label}`;
            }

            if (index === this.selectionIndex) {
                return `> ${label} <`; // Highlight
            }
            return `  ${label}  `;
        }).join('\n');

        this.menuText.text = textStr;

        // Visual polish: Green text for tests, White for main menu
        this.menuText.style.fill = (this.menuState === STATE_TEST) ? '#00FF00' : '#FFFFFF';
    }

    destroy() {
        this.app.stage.removeChild(this.container);
        this.container.destroy({ children: true });
    }
};
