import Phaser from 'phaser';
import { GRID } from '../../core/Constants';

export const AbilityTargetingSystem = (entities, inputManager) => {
    const player = entities.find(e => e.components.PlayerTag);
    if (!player || !player.components.AbilityState || !player.components.Rotation) return;

    const state = player.components.AbilityState;
    const pPos = player.components.Position;
    const pRot = player.components.Rotation.angle;

    // --- CHANGED: Removed X button logic ---
    const holdY = inputManager.isButtonDown('Y');
    const prevArmed = state.armed;

    if (holdY) state.armed = 'Y';
    else state.armed = null;

    if (prevArmed !== state.armed) {
        state.targetIndex = 0;
        state.currentTargetId = null;
    }

    if (!state.armed) return;

    // 2. Filter Configuration
    const config = {
        'Y': { types: ['C'], range: 1000 * GRID.UNIT }
    }[state.armed];

    if (!config) return;

    // 3. The 90-Degree Cone Filter
    const FOV = Math.PI / 2;
    const HALF_FOV = FOV / 2;

    const validTargets = entities.filter(e => {
        if (e === player || !e.components.EntityType) return false;

        const typeMatch = config.types.includes(e.components.EntityType.value);
        const dist = Phaser.Math.Distance.Between(pPos.x, pPos.y, e.components.Position.x, e.components.Position.y);

        if (!typeMatch || dist > config.range) return false;

        const angleToTarget = Phaser.Math.Angle.Between(pPos.x, pPos.y, e.components.Position.x, e.components.Position.y);
        const angleDiff = Phaser.Math.Angle.Wrap(angleToTarget - pRot);

        return Math.abs(angleDiff) <= HALF_FOV;
    }).sort((a, b) => {
        const d1 = Phaser.Math.Distance.Between(pPos.x, pPos.y, a.components.Position.x, a.components.Position.y);
        const d2 = Phaser.Math.Distance.Between(pPos.x, pPos.y, b.components.Position.x, b.components.Position.y);
        const r1 = Math.floor(d1 / 50);
        const r2 = Math.floor(d2 / 50);
        if (r1 !== r2) return r1 - r2;
        const a1 = Math.atan2(a.components.Position.y - pPos.y, a.components.Position.x - pPos.x);
        const a2 = Math.atan2(b.components.Position.y - pPos.y, b.components.Position.x - pPos.x);
        return a1 - a2;
    });

    state.potentialTargets = validTargets.map(t => t.id);
    const count = state.potentialTargets.length;

    // 4. Cycling Logic
    if (count > 0) {
        if (inputManager.getAction('NEXT_TARGET')) {
            state.targetIndex = (state.targetIndex + 1) % count;
        } else if (inputManager.getAction('PREV_TARGET')) {
            state.targetIndex = (state.targetIndex - 1 + count) % count;
        }
        state.currentTargetId = state.potentialTargets[state.targetIndex];
    } else {
        state.currentTargetId = null;
    }
};

// src/ecs/systems/AttachmentSystem.js
import Phaser from 'phaser';

/**
 * AttachmentSystem
 * Handles the 2-frame dwell logic for snapping the player to platforms.
 * Works in tandem with SlingshotSystem's raycast exit to allow re-attachment.
 */
export const AttachmentSystem = (entities) => {
    const player = entities.find(e => e.components.PlayerTag);
    // If no player or already attached, we don't need to check for new attachments
    if (!player || player.components.Parent) return;

    const pPos = player.components.Position;
    const sling = player.components.SlingshotJump;

    // 1. Hard Floor check (Logical Floor is 0)
    // If the player is on the ground, reset the dwell timer
    if (pPos.y >= -2) {
        if (sling) {
            sling.dwellFrames = 0;
            sling.potentialPlatformId = null;
        }
        return;
    }

    const platforms = entities.filter(e => e.components.PlatformTag && e.components.Collider);
    const catchBuffer = 15; // Extra padding to make grabbing feel "generous"
    let currentlyOverId = null;

    // 2. Collision Detection Pass
    for (const plat of platforms) {
        // NOTE: lastPlatformId check removed to allow re-attachment to the same platform

        const platPos = plat.components.Position;
        const col = plat.components.Collider;
        let isInside = false;

        // Circle Collider Logic
        if (col.type === 'circle') {
            const dist = Phaser.Math.Distance.Between(pPos.x, pPos.y, platPos.x, platPos.y);
            isInside = dist <= (col.radius + catchBuffer);
        }
        // Rectangle/Box Collider Logic
        else {
            const hw = (col.width / 2) + catchBuffer;
            const hh = (col.height / 2) + catchBuffer;
            isInside = (
                pPos.x >= platPos.x - hw && pPos.x <= platPos.x + hw &&
                pPos.y >= platPos.y - hh && pPos.y <= platPos.y + hh
            );
        }

        if (isInside) {
            currentlyOverId = plat.id;
            break; // Found a platform, stop looking
        }
    }

    // 3. The 2-Frame Dwell Verification
    // This prevents "flicker" attachments and ensures the player intended to land
    if (currentlyOverId) {
        if (sling.potentialPlatformId === currentlyOverId) {
            sling.dwellFrames++;
        } else {
            // First frame overlapping this specific platform
            sling.potentialPlatformId = currentlyOverId;
            sling.dwellFrames = 1;
        }

        // Trigger the "Snap" on the second consistent frame
        if (sling.dwellFrames >= 2) {
            const plat = entities.find(e => e.id === currentlyOverId);

            // Create the Parent component to lock the player to the platform
            player.components.Parent = {
                _type: 'Parent',
                id: currentlyOverId,
                // Store relative distance so the player stays at the same spot on the platform
                offset: {
                    x: pPos.x - plat.components.Position.x,
                    y: pPos.y - plat.components.Position.y
                }
            };

            // Kill momentum upon attachment
            if (player.components.Velocity) {
                player.components.Velocity.x = 0;
                player.components.Velocity.y = 0;
            }

            // Reset Slingshot State for a new jump
            sling.canJump = true;
            sling.dwellFrames = 0;
            sling.potentialPlatformId = null;

            console.log(`[ATTACH] Snapped to Platform: ${currentlyOverId}`);
        }
    } else {
        // Not overlapping any platform; clear the dwell progress
        if (sling) {
            sling.dwellFrames = 0;
            sling.potentialPlatformId = null;
        }
    }
};

export const CameraSystem = (entities, scene) => {
    const cam = scene.cameras.main;
    const player = entities.find(e => e.components.PlayerTag && e.components.Position);

    // 1. No Player? Look at 0,0
    if (!player) {
        cam.centerOn(0, 0); // Or pan smoothly: cam.pan(0, 0, 1000);
        return;
    }

    const pPos = player.components.Position;
    let targetX = pPos.x;
    let targetY = pPos.y;

    // 2. Check for Active Target (Midpoint Logic)
    const abilityState = player.components.AbilityState;
    if (abilityState && abilityState.currentTargetId) {
        const target = entities.find(e => e.id === abilityState.currentTargetId);

        if (target && target.components.Position) {
            const tPos = target.components.Position;
            // Calculate Midpoint
            targetX = (pPos.x + tPos.x) / 2;
            targetY = (pPos.y + tPos.y) / 2;
        }
    }

    // 3. Move Camera
    // Lerp 0.1 gives a "spring" snap. Set to 1.0 for instant 1-frame snap.
    const lerp = 0.1;

    // Current camera center
    const currentX = cam.scrollX + cam.width / 2;
    const currentY = cam.scrollY + cam.height / 2;

    // Interpolate
    const newX = currentX + (targetX - currentX) * lerp;
    const newY = currentY + (targetY - currentY) * lerp;

    cam.centerOn(newX, newY);
};

export const DebugLayerSystem = (entities, scene) => {
    // Safety check: ensure scene exists
    if (!scene || !scene.add) return;

    // 1. Create or Re-create if missing/destroyed
    if (!scene.debugText || !scene.debugText.active) {
        scene.debugText = scene.add.text(10, 10, '', {
            font: '16px monospace',
            fill: '#00ff00',
            stroke: '#000000',
            strokeThickness: 3,
            backgroundColor: '#00000088'
        });

        scene.debugText.setScrollFactor(0).setDepth(2000);
    }

    const cam = scene.cameras.main;
    const uiScale = 1 / cam.zoom;

    scene.debugText.setScale(uiScale);
    scene.debugText.setPosition(10 * uiScale, 10 * uiScale);

    const player = entities.find(e => e.components.PlayerTag);
    let debugInfo = `SCENE: ${scene.scene.key}\n`;

    if (player) {
        const { Position: pos, Parent: parent } = player.components;
        // Safety check for components
        if (pos) {
            debugInfo += `POS: ${Math.round(pos.x)}, ${Math.round(pos.y)}\n`;
        }

        if (parent) {
            debugInfo += `STATUS: ATTACHED\n`;
            debugInfo += `PARENT: ${parent.id}\n`;
            debugInfo += `OFFSET: ${Math.round(parent.offset.x)}, ${Math.round(parent.offset.y)}\n`;
        } else {
            debugInfo += `STATUS: FREE\n`;
        }
    }

    // 2. Only update text if the object has a valid texture frame
    if (scene.debugText.frame) {
        scene.debugText.setText(debugInfo);
    }
};

import { RenderFactory } from './RenderFactory';

export const LayeredRenderSystem = (entities, scene) => {
    // 1. Find Player's View Layers
    const player = entities.find(e => e.components.PlayerTag);
    const viewLayers = player && player.components.ViewLayers
        ? player.components.ViewLayers.layers
        : ['BASE'];

    entities.forEach(entity => {
        const { Position: pos, Rotation: rot, Renderable: renderable } = entity.components;

        if (!renderable) return;

        const definitions = renderable.definitions;
        const activeGraphics = renderable.activeVisuals; // Holds instances of BaseGraphic

        // 2. Iterate over all defined layers for this entity
        Object.keys(definitions).forEach(layerKey => {
            const config = definitions[layerKey];

            // Logic: Is this layer visible to the player?
            const isVisible = viewLayers.some(view => layerKey === view || layerKey.startsWith(view + '_'));

            if (isVisible) {
                // CASE A: Should be visible
                if (!activeGraphics[layerKey]) {
                    // Create Graphic Instance
                    activeGraphics[layerKey] = RenderFactory.createGraphic(scene, { ...config, layer: layerKey });
                }

                // Delegate Update to the Graphic Instance
                activeGraphics[layerKey].update(pos, rot, config);
                activeGraphics[layerKey].setVisible(true);

            } else {
                // CASE B: Should NOT be visible
                if (activeGraphics[layerKey]) {
                    activeGraphics[layerKey].setVisible(false);
                }
            }
        });

        // 3. Cleanup "Stale" Graphics
        Object.keys(activeGraphics).forEach(activeKey => {
            if (!definitions[activeKey]) {
                activeGraphics[activeKey].destroy(); // The Graphic class cleans up its Phaser object
                delete activeGraphics[activeKey];
            }
        });
    });
};

import Phaser from 'phaser';
import { GRID, PHYSICS } from '../../core/Constants';

export const MovementSystem = (entities, inputManager, dt) => {
    if (!inputManager) return;
    const move = inputManager.getMovement();
    const walkSpeed = GRID.SPEED * GRID.UNIT * 2;

    entities.forEach(entity => {
        const { Position: pos, Velocity: vel, Parent: parent, Rotation: rot, SlingshotJump: sling } = entity.components;
        if (!pos || !rot) return;

        // --- 1. ROTATION LOGIC (PRIORITY: JUMP VECTOR -> MOVEMENT) ---
        if (sling && sling.isCharging && sling.vector && (sling.vector.x !== 0 || sling.vector.y !== 0)) {
            rot.angle = Math.atan2(-sling.vector.y, -sling.vector.x);
        }
        else if (Math.abs(move.x) > 0.1 || Math.abs(move.y) > 0.1) {
            rot.angle = Math.atan2(move.y, move.x);
        }
        else if (!parent && vel && (Math.abs(vel.x) > 10 || Math.abs(vel.y) > 10)) {
            rot.angle = Math.atan2(vel.y, vel.x);
        }

        // --- 2. POSITION LOGIC ---
        if (parent) {
            // Player is attached to a platform
            parent.offset.x += (move.x * walkSpeed * dt);
            parent.offset.y += (move.y * walkSpeed * dt);

            const plat = entities.find(e => e.id === parent.id);
            if (plat) {
                const col = plat.components.Collider;
                const exitBuffer = 25;
                let exited = false;

                if (col.type === 'circle') {
                    const distSq = (parent.offset.x ** 2) + (parent.offset.y ** 2);
                    const limit = col.radius + exitBuffer;
                    if (distSq > limit * limit) exited = true;
                } else {
                    const hw = col.width / 2 + exitBuffer;
                    const hh = col.height / 2 + exitBuffer;
                    if (Math.abs(parent.offset.x) > hw || Math.abs(parent.offset.y) > hh) exited = true;
                }

                if (exited) {
                    // (Optional: remove sling.lastPlatformId = parent.id if you want instant re-grabbing)
                    delete entity.components.Parent;
                }
            }
        } else if (vel) {
            // Player is in freefall or dangling
            const isGrounded = pos.y >= 0;

            if (isGrounded) {
                // Walking on the hard floor (y >= 0)
                vel.x += (move.x * walkSpeed * 10) * dt;
                vel.x *= 0.85;
            } else {
                // AIR CONTROL LOCKOUT
                // We no longer apply move.x here. The Left Stick is ignored.
                // Physics (Gravity & WebSystem) are in full control!
                vel.x *= 0.95; // Standard air friction
            }

            // Always apply gravity when not parented
            vel.y += (PHYSICS.GRAVITY || 980) * dt;
            vel.y *= 0.99;
        }
    });
};

// src/ecs/systems/PhysicsSystem.js
export const PhysicsSystem = (entities, delta) => {
    const dt = delta / 1000;
    const FLOOR_Y = 0;

    entities.forEach(entity => {
        const { Position: pos, Velocity: vel, Parent: parent } = entity.components;
        if (!pos) return;

        // --- ATTACHED SYNC ---
        if (parent) {
            const platform = entities.find(e => e.id === parent.id);
            if (platform && platform.components.Position) {
                // Hard-lock the world position to the platform + offset
                pos.x = platform.components.Position.x + parent.offset.x;
                pos.y = platform.components.Position.y + parent.offset.y;
            }
            if (vel) { vel.x = 0; vel.y = 0; }
            return;
        }

        // --- FREE MOVEMENT ---
        if (vel) {
            pos.x += vel.x * dt;
            pos.y += vel.y * dt;

            if (entity.components.PlayerTag) {
                const sling = entity.components.SlingshotJump;
                // Floor Collision
                if (pos.y >= FLOOR_Y) {
                    pos.y = FLOOR_Y;
                    if (vel.y > 0) vel.y = 0;
                    if (sling) {
                        sling.canJump = true;
                        sling.lastPlatformId = null;
                    }
                } else {
                    // Air state
                    if (sling && !sling.isCharging && pos.y < -5) {
                        sling.canJump = false;
                    }
                }
            }
        }
    });
};

// src/ecs/systems/RaycastPlatformSystem.js
import Phaser from 'phaser';

/**
 * RaycastPlatformSystem
 * Utility for finding the intersection point between a ray (line) 
 * and platform colliders in the ECS.
 */
export const RaycastPlatformSystem = {
    /**
     * Finds the closest platform intersected by a ray.
     * @param {Array} entities - List of ECS entities
     * @param {Object} origin - {x, y} start of the ray
     * @param {Object} direction - Normalized {x, y} direction
     * @param {number} maxRange - Max distance to check
     */
    findTarget: (entities, origin, direction, maxRange = 1000) => {
        const platforms = entities.filter(e => e.components.PlatformTag && e.components.Collider);

        // Define the line segment for the ray
        const ray = new Phaser.Geom.Line(
            origin.x,
            origin.y,
            origin.x + direction.x * maxRange,
            origin.y + direction.y * maxRange
        );

        let closestHit = null;
        let minDist = Infinity;

        platforms.forEach(plat => {
            const pPos = plat.components.Position;
            const col = plat.components.Collider;
            let intersectionPoints = [];

            // 1. Circle Intersection Logic
            if (col.type === 'circle') {
                const circle = new Phaser.Geom.Circle(pPos.x, pPos.y, col.radius);
                intersectionPoints = Phaser.Geom.Intersects.GetLineToCircle(ray, circle);
            }
            // 2. Rectangle Intersection Logic
            else {
                const rect = new Phaser.Geom.Rectangle(
                    pPos.x - col.width / 2,
                    pPos.y - col.height / 2,
                    col.width,
                    col.height
                );
                intersectionPoints = Phaser.Geom.Intersects.GetLineToRectangle(ray, rect);
            }

            // 3. Select the closest point if multiple intersections occurred
            intersectionPoints.forEach(pt => {
                const d = Phaser.Math.Distance.Between(origin.x, origin.y, pt.x, pt.y);
                if (d < minDist) {
                    minDist = d;
                    closestHit = {
                        entity: plat,
                        point: pt,
                        // Relative offset used for snapping/parenting
                        offset: { x: pt.x - pPos.x, y: pt.y - pPos.y }
                    };
                }
            });
        });

        return closestHit;
    }
};

export const RenderSystem = (entities, scene) => {
    if (!scene.gfx) {
        // High base depth for the ECS graphics layer
        scene.gfx = scene.add.graphics().setDepth(1000);
    }
    scene.gfx.clear();

    // 1. Sort entities so Player is always drawn last (on top)
    const sortedEntities = [...entities].sort((a, b) => {
        const aIsPlayer = a.components.PlayerTag ? 1 : 0;
        const bIsPlayer = b.components.PlayerTag ? 1 : 0;
        return aIsPlayer - bIsPlayer;
    });

    sortedEntities.forEach(entity => {
        const { Position: pos, Graphic: gfx, Rotation: rot, PlayerTag: isPlayer, Collider: col } = entity.components;

        if (!pos || !gfx || gfx.texture === 'none') return;

        const color = gfx.color || 0xffffff;
        const angle = (rot && !isNaN(rot.angle)) ? rot.angle : 0;

        scene.gfx.save();
        scene.gfx.translateCanvas(pos.x, pos.y);
        scene.gfx.rotateCanvas(angle);

        // 2. Set alpha/style
        // We can make platforms slightly transparent so the player is even clearer
        const alpha = isPlayer ? 1.0 : 0.8;
        scene.gfx.fillStyle(color, alpha);
        scene.gfx.lineStyle(2, 0xffffff, isPlayer ? 1.0 : 0.5);

        if (isPlayer) {
            const size = gfx.size || 12;
            scene.gfx.beginPath();
            scene.gfx.moveTo(size, 0);
            scene.gfx.lineTo(-size, -size);
            scene.gfx.lineTo(-size, size);
            scene.gfx.closePath();
            scene.gfx.fillPath();
            scene.gfx.strokePath();
        } else {
            if (gfx.texture === 'circle') {
                const r = col ? col.radius : (gfx.size || 8);
                scene.gfx.fillCircle(0, 0, r);
                scene.gfx.strokeCircle(0, 0, r);
            } else {
                const w = col ? col.width : (gfx.size ? gfx.size * 2 : 12);
                const h = col ? col.height : (gfx.size ? gfx.size * 2 : 12);
                scene.gfx.fillRect(-w / 2, -h / 2, w, h);
                scene.gfx.strokeRect(-w / 2, -h / 2, w, h);
            }
        }
        scene.gfx.restore();
    });
};

export const SlingshotRenderSystem = (entities, scene) => {

        scene.slingGfx.strokePath();

        scene.slingGfx.fillStyle(0x00ffff, 1);
        scene.slingGfx.fillCircle(endX, endY, 5);
    }

// src/ecs/systems/SlingshotSystem.js
import { RaycastPlatformSystem } from './RaycastPlatformSystem';

export const SlingshotSystem = (entities, inputManager) => {
    const player = entities.find(e => e.components.PlayerTag && e.components.SlingshotJump);
    if (!player) return;

    const sling = player.components.SlingshotJump;
    const vel = player.components.Velocity;
    const parent = player.components.Parent;
    const pos = player.components.Position;

    const stick = inputManager.getRightStickRaw();
    const mag = Math.sqrt(stick.x * stick.x + stick.y * stick.y);

    const ARM_THRESHOLD = 0.5;
    const RELEASE_THRESHOLD = 0.25;

    if (mag >= ARM_THRESHOLD) {
        if (!sling.isCharging && (parent || (pos && pos.y >= 0))) {
            sling.isCharging = true;
        }
        if (sling.isCharging) {
            sling.lockedAim = { x: stick.x / mag, y: stick.y / mag };
        }
    }
    else if (sling.isCharging && mag < RELEASE_THRESHOLD) {
        if (sling.lockedAim) {
            const launchDir = { x: -sling.lockedAim.x, y: -sling.lockedAim.y };

            if (parent) {
                // Use Raytracing to find the edge of the current platform
                const hit = RaycastPlatformSystem.findTarget(entities, pos, launchDir, 500);
                if (hit && hit.entity.id === parent.id) {
                    // Projection: Move to edge + 10px buffer
                    pos.x = hit.point.x + (launchDir.x * 10);
                    pos.y = hit.point.y + (launchDir.y * 10);
                }
                delete player.components.Parent;
            }

            if (vel) {
                vel.x = launchDir.x * sling.maxForce;
                vel.y = launchDir.y * sling.maxForce;
            }
        }

        sling.isCharging = false;

import Phaser from 'phaser';

export const WebRenderSystem = (entities, scene) => {
    if (!scene.webGfx) {
        scene.webGfx = scene.add.graphics().setDepth(1100);
    }
    scene.webGfx.clear();
    const cam = scene.cameras.main;
    const thickness = 2 / cam.zoom;

    // --- PART 1: DRAW COMPLETED WEBS (Static Connections) ---
    entities.forEach(e => {
        const conn = e.components.WebConnection;
        if (conn) {
            const fromEnt = entities.find(ent => ent.id === conn.fromId);
            const toEnt = entities.find(ent => ent.id === conn.toId);

            if (fromEnt && toEnt) {
                const x1 = fromEnt.components.Position.x + conn.fromOffset.x;
                const y1 = fromEnt.components.Position.y + conn.fromOffset.y;
                const x2 = toEnt.components.Position.x + conn.toOffset.x;
                const y2 = toEnt.components.Position.y + conn.toOffset.y;

                scene.webGfx.lineStyle(thickness, 0xaaaaaa, 0.8); // Silver/Grey
                scene.webGfx.beginPath();
                scene.webGfx.moveTo(x1, y1);
                scene.webGfx.lineTo(x2, y2);
                scene.webGfx.strokePath();

                scene.webGfx.fillStyle(0xaaaaaa, 1);
                scene.webGfx.fillCircle(x1, y1, 3 / cam.zoom);
                scene.webGfx.fillCircle(x2, y2, 3 / cam.zoom);
            }
        }
    });

    // --- PART 2: DRAW ACTIVE TETHER (With Slack & Momentum) ---
    const player = entities.find(e => e.components.PlayerTag);
    if (!player) return;

    const tether = player.components.WebTether;
    const pPos = player.components.Position;
    const pVel = player.components.Velocity;

    if (tether && pPos) {
        const platform = entities.find(e => e.id === tether.platformId);
        if (platform && platform.components.Position) {
            const anchorX = platform.components.Position.x + tether.anchorOffset.x;
            const anchorY = platform.components.Position.y + tether.anchorOffset.y;

            const dist = Phaser.Math.Distance.Between(pPos.x, pPos.y, anchorX, anchorY);
            const isTaut = dist >= (tether.maxLength - 5);

            scene.webGfx.lineStyle(thickness, isTaut ? 0xffa500 : 0xffffff, isTaut ? 1.0 : 0.6);
            scene.webGfx.beginPath();
            scene.webGfx.moveTo(anchorX, anchorY);

            if (isTaut) {
                // Straight line when stretched
                scene.webGfx.lineTo(pPos.x, pPos.y);
            } else {
                // --- STRICTLY USING lineTo ---
                const segments = 10;
                const slack = (tether.maxLength - dist) * 0.5;

                // Momentum: drag the web in opposite direction of velocity
                let dragX = 0;
                let dragY = 0;
                if (pVel) {
                    dragX = -pVel.x * 0.04;
                    dragY = -pVel.y * 0.04;
                }

                for (let i = 1; i <= segments; i++) {
                    const t = i / segments;

                    const currX = Phaser.Math.Interpolation.Linear([anchorX, pPos.x], t);
                    const currY = Phaser.Math.Interpolation.Linear([anchorY, pPos.y], t);

                    // Arc multiplier keeps ends attached, moves the middle
                    const arc = 4 * t * (1 - t);

                    const offsetX = dragX * arc;
                    const offsetY = (slack + dragY) * arc;

                    scene.webGfx.lineTo(currX + offsetX, currY + offsetY);
                }
            }

            scene.webGfx.strokePath();

            // Draw the anchor point dot
            scene.webGfx.fillStyle(isTaut ? 0xffa500 : 0xffffff, 1);
            scene.webGfx.fillCircle(anchorX, anchorY, 4 / cam.zoom);
        }
    }
};

import Phaser from 'phaser';
import { WebTether, WebConnection, Parent } from '../components/index';

export const WebSystem = (entities, inputManager) => {
    const player = entities.find(e => e.components.PlayerTag);
    if (!player) return;

    const tether = player.components.WebTether;
    const currentParent = player.components.Parent;
    const pPos = player.components.Position;
    const pVel = player.components.Velocity;

    const MAX_SPOOL_LENGTH = 1000; // Updated to match your component
    const MIN_SPOOL_LENGTH = 10;

    // --- 1. WINCH & SPRING PHYSICS ---
    if (tether) {
        const plat = entities.find(e => e.id === tether.platformId);
        if (plat && plat.components.Position) {
            const anchorX = plat.components.Position.x + tether.anchorOffset.x;
            const anchorY = plat.components.Position.y + tether.anchorOffset.y;

            const dist = Phaser.Math.Distance.Between(pPos.x, pPos.y, anchorX, anchorY);
            tether.currentLength = dist;

            const lb = inputManager.isButtonDown('SHORTEN');
            const rb = inputManager.isButtonDown('LENGTHEN');
            const winchSpeed = 4;

            // Spool / Unspool (Change Max Length)
            if (lb) tether.maxLength = Math.max(MIN_SPOOL_LENGTH, tether.maxLength - winchSpeed);
            if (rb) tether.maxLength = Math.min(MAX_SPOOL_LENGTH, tether.maxLength + winchSpeed);

            // A. IF STANDING ON PLATFORM (Parented)
            if (currentParent) {
                const dx = currentParent.offset.x - tether.anchorOffset.x;
                const dy = currentParent.offset.y - tether.anchorOffset.y;
                const offsetDist = Math.sqrt(dx * dx + dy * dy);

                if (offsetDist > tether.maxLength) {
                    const diff = offsetDist - tether.maxLength;
                    const dirX = dx / offsetDist;
                    const dirY = dy / offsetDist;

                    currentParent.offset.x -= dirX * diff;
                    currentParent.offset.y -= dirY * diff;
                }
            }

            // B. IF DANGLING IN AIR (Free-Hanging with Spring Physics)
            else if (!currentParent && pVel) {
                if (dist > tether.maxLength) {
                    // 1. Find the direction vector from Anchor TO Player
                    const nx = (pPos.x - anchorX) / dist;
                    const ny = (pPos.y - anchorY) / dist;

                    // 2. HARD POSITION CONSTRAINT: You cannot stretch the rope
                    pPos.x = anchorX + (nx * tether.maxLength);
                    pPos.y = anchorY + (ny * tether.maxLength);

                    // 3. TANGENT VELOCITY PROJECTION
                    // Find exactly how much of the current velocity is pushing OUTWARD
                    const outwardVel = (pVel.x * nx) + (pVel.y * ny);

                    // If moving outward, subtract only that outward force.
                    // This seamlessly preserves perpendicular momentum, creating a perfect swing arc.
                    if (outwardVel > 0) {
                        pVel.x -= outwardVel * nx;
                        pVel.y -= outwardVel * ny;
                    }

                    // Apply a very light air resistance to the swing (do not use massive damping)
                    pVel.x *= 0.999;
                    pVel.y *= 0.999;
                }
            }
        }

        // --- 2. X & Y BUTTON LOGIC ---
        const xJustPressed = inputManager.isButtonPressed('X');
        const yJustPressed = inputManager.isButtonPressed('Y');

        // ATTACH
        if ((xJustPressed || yJustPressed) && !tether && currentParent) {
            const dx = currentParent.offset.x;
            const dy = currentParent.offset.y;
            let initialDist = Math.sqrt(dx * dx + dy * dy);
            initialDist = Phaser.Math.Clamp(initialDist, MIN_SPOOL_LENGTH, MAX_SPOOL_LENGTH);

            // 1. Create the component using your exact setup
            player.components.WebTether = WebTether(currentParent.id, { x: 0, y: 0 });

            // 2. Immediately overwrite the hardcoded 1000 so it starts perfectly taut
            player.components.WebTether.maxLength = initialDist;
            player.components.WebTether.currentLength = initialDist;
            return;
        }

        // DETACH
        if (xJustPressed && tether) {
            delete player.components.WebTether;
            return;
        }

        // CONNECT
        if (yJustPressed && tether && currentParent && tether.platformId !== currentParent.id) {
            const exists = entities.some(e => {
                const c = e.components.WebConnection;
                return c && (
                    (c.fromId === tether.platformId && c.toId === currentParent.id) ||
                    (c.fromId === currentParent.id && c.toId === tether.platformId)
                );
            });

            if (!exists) {
                entities.push({
                    id: crypto.randomUUID(),
                    components: {
                        WebConnection: WebConnection(
                            tether.platformId,
                            { ...tether.anchorOffset },
                            currentParent.id,
                            { x: 0, y: 0 }
                        )
                    }
                });
            }
            delete player.components.WebTether;
        }

        // LASSO (Fallback for X if not parented)
        if (xJustPressed && !tether && !currentParent) {
            const platforms = entities.filter(e => e.components.PlatformTag && e.components.Collider);
            const target = platforms.find(plat => {
                const dist = Phaser.Math.Distance.Between(pPos.x, pPos.y, plat.components.Position.x, plat.components.Position.y);
                return dist <= (plat.components.Collider.radius || 50) + 20;
            });
            if (target) {
                player.components.Parent = Parent(target.id, {
                    x: pPos.x - target.components.Position.x,
                    y: pPos.y - target.components.Position.y
                });
            }
        }
    };
