import Phaser from 'phaser';

export class InputManager {
    constructor(scene) {
        this.scene = scene;
        this.cursors = scene.input.keyboard.createCursorKeys();

        this.keys = scene.input.keyboard.addKeys({
            enter: Phaser.Input.Keyboard.KeyCodes.ENTER,
            esc: Phaser.Input.Keyboard.KeyCodes.ESC,
            x: Phaser.Input.Keyboard.KeyCodes.X,
            y: Phaser.Input.Keyboard.KeyCodes.Y,
            b: Phaser.Input.Keyboard.KeyCodes.B,
            tab: Phaser.Input.Keyboard.KeyCodes.TAB,
            shorten: Phaser.Input.Keyboard.KeyCodes.Q, // LB equivalent
            lengthen: Phaser.Input.Keyboard.KeyCodes.E, // RB equivalent
            // --- NEW: WASD for twin-stick keyboard fallback ---
            w: Phaser.Input.Keyboard.KeyCodes.W,
            a: Phaser.Input.Keyboard.KeyCodes.A,
            s: Phaser.Input.Keyboard.KeyCodes.S,
            d: Phaser.Input.Keyboard.KeyCodes.D
        });

        this.pad = null;
        this.prevPadState = {};

        if (this.scene.input.gamepad.total > 0) {
            this.pad = this.scene.input.gamepad.getPad(0);
        }

        this.scene.input.gamepad.on('connected', (pad) => {
            this.pad = pad;
        });
    }

    isButtonDown(action) {
        if (action === 'X' && this.keys.x.isDown) return true;
        if (action === 'Y' && this.keys.y.isDown) return true;
        if (action === 'B' && (this.keys.b.isDown || this.keys.esc.isDown)) return true;

        if (action === 'SHORTEN') {
            if (this.keys.shorten.isDown) return true;
            if (this.pad && (this.pad.L1 || this.pad.buttons[4].pressed)) return true;
        }
        if (action === 'LENGTHEN') {
            if (this.keys.lengthen.isDown) return true;
            if (this.pad && (this.pad.R1 || this.pad.buttons[5].pressed)) return true;
        }

        if (this.pad) {
            if (action === 'X' && (this.pad.X || this.pad.buttons[2].pressed)) return true;
            if (action === 'Y' && (this.pad.Y || this.pad.buttons[3].pressed)) return true;
            if (action === 'B' && (this.pad.B || this.pad.buttons[1].pressed)) return true;
        }
        return false;
    }

    getAction(action) {
        const justDown = (key) => Phaser.Input.Keyboard.JustDown(key);

        if (action === 'UP') return justDown(this.cursors.up) || this.isButtonPressed('up');
        if (action === 'DOWN') return justDown(this.cursors.down) || this.isButtonPressed('down');
        if (action === 'SELECT') return justDown(this.keys.enter) || this.isButtonPressed('A');
        if (action === 'BACK') return justDown(this.keys.esc) || this.isButtonPressed('B');

        if (action === 'NEXT_TARGET') return justDown(this.keys.lengthen) || this.isButtonPressed('R1');
        if (action === 'PREV_TARGET') return justDown(this.keys.shorten) || this.isButtonPressed('L1');

        return false;
    }

    isButtonPressed(button) {
        if (!this.pad) return false;

        let isDown = false;

        if (button === 'up') isDown = this.pad.up;
        else if (button === 'down') isDown = this.pad.down;
        else if (button === 'left') isDown = this.pad.left;
        else if (button === 'right') isDown = this.pad.right;
        else if (button === 'A') isDown = this.pad.A;
        else if (button === 'B') isDown = this.pad.B;
        else if (button === 'X') isDown = this.pad.X;
        else if (button === 'Y') isDown = this.pad.Y;
        else if (button === 'L1') isDown = this.pad.L1;
        else if (button === 'R1') isDown = this.pad.R1;
        else if (this.pad[button]) isDown = this.pad[button];

        const wasDown = this.prevPadState[button] || false;
        this.prevPadState[button] = isDown;

        return isDown && !wasDown;
    }

    // --- UPDATED: Left Stick / WASD ---
    getMovement() {
        let x = 0, y = 0;

        if (this.keys.a.isDown) x = -1;
        else if (this.keys.d.isDown) x = 1;
        if (this.keys.w.isDown) y = -1;
        else if (this.keys.s.isDown) y = 1;

        if (this.pad) {
            const t = 0.15;
            if (Math.abs(this.pad.leftStick.x) > t) x = this.pad.leftStick.x;
            if (Math.abs(this.pad.leftStick.y) > t) y = this.pad.leftStick.y;
        }
        return { x, y };
    }

    // --- NEW: Right Stick / Arrows ---
    getRightStickRaw() {
        let x = 0, y = 0;

        if (this.pad) {
            x = this.pad.rightStick.x;
            y = this.pad.rightStick.y;
        } else {
            if (this.cursors.left.isDown) x = -1;
            else if (this.cursors.right.isDown) x = 1;
            if (this.cursors.up.isDown) y = -1;
            else if (this.cursors.down.isDown) y = 1;
        }

        return { x, y };
    }
}

export class UIManager {
    constructor() {
        if (typeof document === 'undefined') return;

        // Prevent duplicate overlays
        if (document.getElementById('game-ui-overlay')) {
            this.overlay = document.getElementById('game-ui-overlay');
            return;
        }

        this.overlay = document.createElement('div');
        this.overlay.id = 'game-ui-overlay';
        this.overlay.style.cssText = `
            position: absolute; top: 0; left: 0; width: 100%; height: 100%;
            pointer-events: none; z-index: 1000; font-family: monospace; color: #fff;
            overflow: hidden;
        `;
        document.body.appendChild(this.overlay);
    }

    showTitle(title = "SCENE", sub = "") {
        const el = document.createElement('div');
        el.style.cssText = `
            position: absolute; top: 50%; left: 50%; transform: translate(-50%, -50%);
            text-align: center; opacity: 0; transition: opacity 0.5s ease-in-out;
        `;

        el.innerHTML = `
            <h1 style="color: #8b008b; font-size: 64px; margin: 0; font-weight: bold; text-shadow: 2px 2px #000;">
                ${title.toString().toUpperCase()}
            </h1>
            <p style="color: #ff69b4; font-size: 24px; margin: 10px 0 0 0; text-shadow: 1px 1px #000;">
                ${sub.toString()}
            </p>
        `;

        this.overlay.appendChild(el);
        requestAnimationFrame(() => { el.style.opacity = '1'; });

        setTimeout(() => {
            el.style.opacity = '0';
            setTimeout(() => el.remove(), 1000);
        }, 3000);
    }

    updateDebug(playerData, stateData) {
        let debugEl = document.getElementById('debug-console');
        if (!debugEl) {
            debugEl = document.createElement('div');
            debugEl.id = 'debug-console';
            debugEl.style.cssText = `
                position: absolute; top: 0; left: 0; width: 340px;
                background: rgba(0,0,0,0.85); padding: 20px;
                display: flex; flex-direction: column; align-items: flex-start;
                font-size: 13px; color: #0f0; visibility: visible;
                border-bottom: 1px solid #0f0; border-right: 1px solid #0f0;
                pointer-events: auto;
            `;
            this.overlay.appendChild(debugEl);
        }

        const sling = stateData.sling || {};
        const raw = sling.rawStick || { x: 0, y: 0 };
        const mag = Math.sqrt(raw.x * raw.x + raw.y * raw.y);

        // Visualize the Deadbolt State
        const isLocked = mag < 0.5 && sling.isCharging;
        const stateTag = isLocked ? '<span style="color:#ff0000">LOCKED</span>' :
            mag >= 0.5 ? '<span style="color:#00ff00">AIMING</span>' : 'IDLE';

        debugEl.innerHTML = `
            <div style="margin-bottom: 10px; border-bottom: 1px solid #444; width: 100%; padding-bottom: 5px;">
                <strong style="color: #0ff;">[ PLAYER ]</strong><br>
                POS: ${playerData.x}, ${playerData.y}<br>
                STATUS: ${stateData.status}
            </div>
            <div style="color: #ff8400;">
                <strong style="color: #fff;">[ SLINGSHOT ]</strong><br>
                STATE: ${stateTag}<br>
                RAW MAG: ${mag.toFixed(3)}<br>
                LOCKED VEC: ${sling.vecX.toFixed(2)}, ${sling.vecY.toFixed(2)}<br>
                <br>
                <strong style="color: #fff;">[ DIAGNOSTICS ]</strong><br>
                STICK X: ${raw.x.toFixed(3)}<br>
                STICK Y: ${raw.y.toFixed(3)}
            </div>
        `;
    }

    toggleDebug() {
        const el = document.getElementById('debug-console');
        if (el) {
            el.style.visibility = (el.style.visibility === 'hidden') ? 'visible' : 'hidden';
        }
    }
}

import Phaser from 'phaser';
import { GRID } from '../../core/Constants';

export const AbilityTargetingSystem = (entities, inputManager) => {
    const player = entities.find(e => e.components.PlayerTag);
    if (!player || !player.components.AbilityState || !player.components.Rotation) return;

    const state = player.components.AbilityState;
    const pPos = player.components.Position;
    const pRot = player.components.Rotation.angle;

    // --- CHANGED: Removed X button logic ---
    const holdY = inputManager.isButtonDown('Y');
    const prevArmed = state.armed;

    if (holdY) state.armed = 'Y';
    else state.armed = null;

    if (prevArmed !== state.armed) {
        state.targetIndex = 0;
        state.currentTargetId = null;
    }

    if (!state.armed) return;

    // 2. Filter Configuration
    const config = {
        'Y': { types: ['C'], range: 1000 * GRID.UNIT }
    }[state.armed];

    if (!config) return;

    // 3. The 90-Degree Cone Filter
    const FOV = Math.PI / 2;
    const HALF_FOV = FOV / 2;

    const validTargets = entities.filter(e => {
        if (e === player || !e.components.EntityType) return false;

        const typeMatch = config.types.includes(e.components.EntityType.value);
        const dist = Phaser.Math.Distance.Between(pPos.x, pPos.y, e.components.Position.x, e.components.Position.y);

        if (!typeMatch || dist > config.range) return false;

        const angleToTarget = Phaser.Math.Angle.Between(pPos.x, pPos.y, e.components.Position.x, e.components.Position.y);
        const angleDiff = Phaser.Math.Angle.Wrap(angleToTarget - pRot);

        return Math.abs(angleDiff) <= HALF_FOV;
    }).sort((a, b) => {
        const d1 = Phaser.Math.Distance.Between(pPos.x, pPos.y, a.components.Position.x, a.components.Position.y);
        const d2 = Phaser.Math.Distance.Between(pPos.x, pPos.y, b.components.Position.x, b.components.Position.y);
        const r1 = Math.floor(d1 / 50);
        const r2 = Math.floor(d2 / 50);
        if (r1 !== r2) return r1 - r2;
        const a1 = Math.atan2(a.components.Position.y - pPos.y, a.components.Position.x - pPos.x);
        const a2 = Math.atan2(b.components.Position.y - pPos.y, b.components.Position.x - pPos.x);
        return a1 - a2;
    });

    state.potentialTargets = validTargets.map(t => t.id);
    const count = state.potentialTargets.length;

    // 4. Cycling Logic
    if (count > 0) {
        if (inputManager.getAction('NEXT_TARGET')) {
            state.targetIndex = (state.targetIndex + 1) % count;
        } else if (inputManager.getAction('PREV_TARGET')) {
            state.targetIndex = (state.targetIndex - 1 + count) % count;
        }
        state.currentTargetId = state.potentialTargets[state.targetIndex];
    } else {
        state.currentTargetId = null;
    }
};

// src/ecs/systems/AttachmentSystem.js
import Phaser from 'phaser';

/**
 * AttachmentSystem
 * Handles the 2-frame dwell logic for snapping the player to platforms.
 * Works in tandem with SlingshotSystem's raycast exit to allow re-attachment.
 */
export const AttachmentSystem = (entities) => {
    const player = entities.find(e => e.components.PlayerTag);
    // If no player or already attached, we don't need to check for new attachments
    if (!player || player.components.Parent) return;

    const pPos = player.components.Position;
    const sling = player.components.SlingshotJump;

    // 1. Hard Floor check (Logical Floor is 0)
    // If the player is on the ground, reset the dwell timer
    if (pPos.y >= -2) {
        if (sling) {
            sling.dwellFrames = 0;
            sling.potentialPlatformId = null;
        }
        return;
    }

    const platforms = entities.filter(e => e.components.PlatformTag && e.components.Collider);
    const catchBuffer = 15; // Extra padding to make grabbing feel "generous"
    let currentlyOverId = null;

    // 2. Collision Detection Pass
    for (const plat of platforms) {
        // NOTE: lastPlatformId check removed to allow re-attachment to the same platform

        const platPos = plat.components.Position;
        const col = plat.components.Collider;
        let isInside = false;

        // Circle Collider Logic
        if (col.type === 'circle') {
            const dist = Phaser.Math.Distance.Between(pPos.x, pPos.y, platPos.x, platPos.y);
            isInside = dist <= (col.radius + catchBuffer);
        }
        // Rectangle/Box Collider Logic
        else {
            const hw = (col.width / 2) + catchBuffer;
            const hh = (col.height / 2) + catchBuffer;
            isInside = (
                pPos.x >= platPos.x - hw && pPos.x <= platPos.x + hw &&
                pPos.y >= platPos.y - hh && pPos.y <= platPos.y + hh
            );
        }

        if (isInside) {
            currentlyOverId = plat.id;
            break; // Found a platform, stop looking
        }
    }

    // 3. The 2-Frame Dwell Verification
    // This prevents "flicker" attachments and ensures the player intended to land
    if (currentlyOverId) {
        if (sling.potentialPlatformId === currentlyOverId) {
            sling.dwellFrames++;
        } else {
            // First frame overlapping this specific platform
            sling.potentialPlatformId = currentlyOverId;
            sling.dwellFrames = 1;
        }

        // Trigger the "Snap" on the second consistent frame
        if (sling.dwellFrames >= 2) {
            const plat = entities.find(e => e.id === currentlyOverId);

            // Create the Parent component to lock the player to the platform
            player.components.Parent = {
                _type: 'Parent',
                id: currentlyOverId,
                // Store relative distance so the player stays at the same spot on the platform
                offset: {
                    x: pPos.x - plat.components.Position.x,
                    y: pPos.y - plat.components.Position.y
                }
            };

            // Kill momentum upon attachment
            if (player.components.Velocity) {
                player.components.Velocity.x = 0;
                player.components.Velocity.y = 0;
            }

            // Reset Slingshot State for a new jump
            sling.canJump = true;
            sling.dwellFrames = 0;
            sling.potentialPlatformId = null;

            console.log(`[ATTACH] Snapped to Platform: ${currentlyOverId}`);
        }
    } else {
        // Not overlapping any platform; clear the dwell progress
        if (sling) {
            sling.dwellFrames = 0;
            sling.potentialPlatformId = null;
        }
    }
};

export const CameraSystem = (entities, scene) => {
    const cam = scene.cameras.main;
    const player = entities.find(e => e.components.PlayerTag && e.components.Position);

    // 1. No Player? Look at 0,0
    if (!player) {
        cam.centerOn(0, 0); // Or pan smoothly: cam.pan(0, 0, 1000);
        return;
    }

    const pPos = player.components.Position;
    let targetX = pPos.x;
    let targetY = pPos.y;

    // 2. Check for Active Target (Midpoint Logic)
    const abilityState = player.components.AbilityState;
    if (abilityState && abilityState.currentTargetId) {
        const target = entities.find(e => e.id === abilityState.currentTargetId);

        if (target && target.components.Position) {
            const tPos = target.components.Position;
            // Calculate Midpoint
            targetX = (pPos.x + tPos.x) / 2;
            targetY = (pPos.y + tPos.y) / 2;
        }
    }

    // 3. Move Camera
    // Lerp 0.1 gives a "spring" snap. Set to 1.0 for instant 1-frame snap.
    const lerp = 0.1;

    // Current camera center
    const currentX = cam.scrollX + cam.width / 2;
    const currentY = cam.scrollY + cam.height / 2;

    // Interpolate
    const newX = currentX + (targetX - currentX) * lerp;
    const newY = currentY + (targetY - currentY) * lerp;

    cam.centerOn(newX, newY);
};

export const DebugLayerSystem = (entities, scene) => {
    // Safety check: ensure scene exists
    if (!scene || !scene.add) return;

    // 1. Create or Re-create if missing/destroyed
    if (!scene.debugText || !scene.debugText.active) {
        scene.debugText = scene.add.text(10, 10, '', {
            font: '16px monospace',
            fill: '#00ff00',
            stroke: '#000000',
            strokeThickness: 3,
            backgroundColor: '#00000088'
        });

        scene.debugText.setScrollFactor(0).setDepth(2000);
    }

    const cam = scene.cameras.main;
    const uiScale = 1 / cam.zoom;

    scene.debugText.setScale(uiScale);
    scene.debugText.setPosition(10 * uiScale, 10 * uiScale);

    const player = entities.find(e => e.components.PlayerTag);
    let debugInfo = `SCENE: ${scene.scene.key}\n`;

    if (player) {
        const { Position: pos, Parent: parent } = player.components;
        // Safety check for components
        if (pos) {
            debugInfo += `POS: ${Math.round(pos.x)}, ${Math.round(pos.y)}\n`;
        }

        if (parent) {
            debugInfo += `STATUS: ATTACHED\n`;
            debugInfo += `PARENT: ${parent.id}\n`;
            debugInfo += `OFFSET: ${Math.round(parent.offset.x)}, ${Math.round(parent.offset.y)}\n`;
        } else {
            debugInfo += `STATUS: FREE\n`;
        }
    }

    // 2. Only update text if the object has a valid texture frame
    if (scene.debugText.frame) {
        scene.debugText.setText(debugInfo);
    }
};

export const LayeredRenderSystem = (entities, scene) => {
    // 1. Find Player's View Layers
    const player = entities.find(e => e.components.PlayerTag);
    const viewLayers = player && player.components.ViewLayers
        ? player.components.ViewLayers.layers
        : ['BASE'];

    entities.forEach(entity => {
        const { Position: pos, Rotation: rot, Renderable: renderable } = entity.components;

        if (!renderable) return;

        const definitions = renderable.definitions;
        const activeGraphics = renderable.activeVisuals; // Holds instances of BaseGraphic

        // 2. Iterate over all defined layers for this entity
        Object.keys(definitions).forEach(layerKey => {
            const config = definitions[layerKey];

            // Logic: Is this layer visible to the player?
            const isVisible = viewLayers.some(view => layerKey === view || layerKey.startsWith(view + '_'));

            if (isVisible) {
                // CASE A: Should be visible
                if (!activeGraphics[layerKey]) {

export const MovementSystem = (entities, inputManager, dt) => {
    if (!inputManager) return;
    const move = inputManager.getMovement();
    const walkSpeed = GRID.SPEED * GRID.UNIT * 2;

    entities.forEach(entity => {
        // ADDED WebTether to the destructured components
        const { Position: pos, Velocity: vel, Parent: parent, Rotation: rot, SlingshotJump: sling, WebTether: tether } = entity.components;
        if (!pos || !rot) return;

        // --- 1. ROTATION LOGIC (PRIORITY: JUMP VECTOR -> MOVEMENT) ---
        if (sling && sling.isCharging && sling.vector && (sling.vector.x !== 0 || sling.vector.y !== 0)) {
            rot.angle = Math.atan2(-sling.vector.y, -sling.vector.x);
        }
        else if (Math.abs(move.x) > 0.1 || Math.abs(move.y) > 0.1) {
            rot.angle = Math.atan2(move.y, move.x);
        }
        else if (!parent && vel && (Math.abs(vel.x) > 10 || Math.abs(vel.y) > 10)) {
            rot.angle = Math.atan2(vel.y, vel.x);
        }

        // --- 2. POSITION LOGIC ---
        if (parent) {
            // Player is attached to a platform
            parent.offset.x += (move.x * walkSpeed * dt);
            parent.offset.y += (move.y * walkSpeed * dt);

            const plat = entities.find(e => e.id === parent.id);
            if (plat) {
                const col = plat.components.Collider;
                const exitBuffer = 25;
                let exited = false;

                if (col.type === 'circle') {
                    const distSq = (parent.offset.x ** 2) + (parent.offset.y ** 2);
                    const limit = col.radius + exitBuffer;
                    if (distSq > limit * limit) exited = true;
                } else {
                    const hw = col.width / 2 + exitBuffer;
                    const hh = col.height / 2 + exitBuffer;
                    if (Math.abs(parent.offset.x) > hw || Math.abs(parent.offset.y) > hh) exited = true;
                }

                if (exited) {
                    if (sling) sling.lastPlatformId = parent.id;
                    delete entity.components.Parent;
                }
            }
        } else if (vel) {
            // Player is in freefall or dangling
            const isGrounded = pos.y >= 0;

            if (isGrounded) {
                // Walking on the hard floor (y >= 0)
                vel.x += (move.x * walkSpeed * 10) * dt;
                vel.x *= 0.85;
            } else {
                // --- THE FIX: TETHER BYPASS ---
                if (!tether) {
                    // Normal Freefall: apply standard slingshot air friction
                    vel.x *= 0.95;
                    vel.y *= 0.99;
                } else {
                    // On a Rope: DO NOT kill the momentum! Let the pendulum swing!
                    // Apply only a micro-fraction of air resistance so the swing lasts.
                    vel.x *= 0.999;
                    vel.y *= 0.999;
                }
            }

            // Always apply gravity when not parented
            vel.y += (PHYSICS.GRAVITY || 980) * dt;
        }
    });

export const PhysicsSystem = (entities, delta) => {
    const dt = delta / 1000;
    const FLOOR_Y = 0;

    entities.forEach(entity => {
        const { Position: pos, Velocity: vel, Parent: parent } = entity.components;
        if (!pos) return;

        // --- ATTACHED SYNC ---
        if (parent) {
            const platform = entities.find(e => e.id === parent.id);
            if (platform && platform.components.Position) {
                // Hard-lock the world position to the platform + offset
                pos.x = platform.components.Position.x + parent.offset.x;
                pos.y = platform.components.Position.y + parent.offset.y;
            }
            if (vel) { vel.x = 0; vel.y = 0; }
            return;
        }

        // --- FREE MOVEMENT ---
        if (vel) {
            pos.x += vel.x * dt;
            pos.y += vel.y * dt;

            if (entity.components.PlayerTag) {
                const sling = entity.components.SlingshotJump;
                // Floor Collision
                if (pos.y >= FLOOR_Y) {
                    pos.y = FLOOR_Y;
                    if (vel.y > 0) vel.y = 0;
                    if (sling) {
                        sling.canJump = true;
                        sling.lastPlatformId = null;
                    }
                } else {
                    // Air state
                    if (sling && !sling.isCharging && pos.y < -5) {
                        sling.canJump = false;
                    }
                }
            }
        }
    });

// src/ecs/systems/RaycastPlatformSystem.js
import Phaser from 'phaser';

/**
 * RaycastPlatformSystem
 * Utility for finding the intersection point between a ray (line) 
 * and platform colliders in the ECS.
 */
export const RaycastPlatformSystem = {
    /**
     * Finds the closest platform intersected by a ray.
     * @param {Array} entities - List of ECS entities
     * @param {Object} origin - {x, y} start of the ray
     * @param {Object} direction - Normalized {x, y} direction
     * @param {number} maxRange - Max distance to check
     */
    findTarget: (entities, origin, direction, maxRange = 1000) => {
        const platforms = entities.filter(e => e.components.PlatformTag && e.components.Collider);

        // Define the line segment for the ray
        const ray = new Phaser.Geom.Line(
            origin.x,
            origin.y,
            origin.x + direction.x * maxRange,
            origin.y + direction.y * maxRange
        );

        let closestHit = null;
        let minDist = Infinity;

        platforms.forEach(plat => {
            const pPos = plat.components.Position;
            const col = plat.components.Collider;
            let intersectionPoints = [];

            // 1. Circle Intersection Logic
            if (col.type === 'circle') {
                const circle = new Phaser.Geom.Circle(pPos.x, pPos.y, col.radius);
                intersectionPoints = Phaser.Geom.Intersects.GetLineToCircle(ray, circle);
            }
            // 2. Rectangle Intersection Logic
            else {
                const rect = new Phaser.Geom.Rectangle(
                    pPos.x - col.width / 2,
                    pPos.y - col.height / 2,
                    col.width,
                    col.height
                );
                intersectionPoints = Phaser.Geom.Intersects.GetLineToRectangle(ray, rect);
            }

            // 3. Select the closest point if multiple intersections occurred
            intersectionPoints.forEach(pt => {
                const d = Phaser.Math.Distance.Between(origin.x, origin.y, pt.x, pt.y);
                if (d < minDist) {
                    minDist = d;
                    closestHit = {
                        entity: plat,
                        point: pt,
                        // Relative offset used for snapping/parenting
                        offset: { x: pt.x - pPos.x, y: pt.y - pPos.y }
                    };
                }
            });
        });

        return closestHit;
    }
};

        }

    else if (sling.isCharging && mag < RELEASE_THRESHOLD) {
        if (sling.lockedAim) {
            const launchDir = { x: -sling.lockedAim.x, y: -sling.lockedAim.y };

                vel.x = launchDir.x * sling.maxForce;

            return c && ((c.fromId === tether.platformId && c.toId === currentParent.id) || (c.fromId === currentParent.id && c.toId === tether.platformId));
        });

import Phaser from 'phaser';
import { BaseGraphic } from './BaseGraphic';

export class ChaosEmitterGraphic extends BaseGraphic {
    create(config) {
        this.gameObject = this.scene.add.circle(0, 0, (config.size || 1) * 10, config.coreColor || 0xffff00, 0.2);
        this.gameObject.setDepth(100);

        this.tracerGfx = this.scene.add.graphics().setDepth(90);
        this.particles = [];
        this.spawnTimer = 0;
        return this.gameObject;
    }

    onUpdate(config) {
        if (!this.gameObject) return;

        const intensity = config.intensity ?? 1.0;
        this.spawnTimer++;

        const spawnRate = config.spawnRate ?? (10 / intensity);
        const maxParticles = config.maxParticles ?? (20 * intensity);

        if (this.spawnTimer > spawnRate && this.particles.length < maxParticles) {
            this.spawnParticle(config);
            this.spawnTimer = 0;
        }

        this.tracerGfx.clear();

        for (let i = this.particles.length - 1; i >= 0; i--) {
            const p = this.particles[i];

            // Chaos Jitter
            p.vx += (Math.random() - 0.5) * (config.jitter ?? 0.8) * intensity;
            p.vy += (Math.random() - 0.5) * (config.jitter ?? 0.8) * intensity;

            const dx = p.x - this.gameObject.x;
            const dy = p.y - this.gameObject.y;
            const dist = Math.sqrt(dx * dx + dy * dy);

            if (dist > 0) {
                const pushForce = (config.push ?? 0.15) * intensity;
                p.vx += (dx / dist) * pushForce;
                p.vy += (dy / dist) * pushForce;

                // ABSORPTION KILL-ZONE: If push is negative, kill particle when it hits center
                if (pushForce < 0 && dist < 8) {
                    p.life = 0;
                }
            }

            p.vx *= (config.friction ?? 0.92);
            p.vy *= (config.friction ?? 0.92);

            p.x += p.vx;
            p.y += p.vy;
            p.life -= 1;

            p.history.push({ x: p.x, y: p.y });
            if (p.history.length > (config.tailLen ?? 15)) p.history.shift();

            this.drawParticle(p, config, intensity);
            if (p.life <= 0) this.particles.splice(i, 1);
        }
    }

    drawParticle(p, config, intensity) {
        const alpha = (p.life / p.maxLife);
        if (p.history.length > 1) {
            this.tracerGfx.lineStyle(config.thickness ?? 1, config.tracerColor ?? 0xffa500, alpha);
            this.tracerGfx.beginPath();
            this.tracerGfx.moveTo(p.history[0].x, p.history[0].y);
            p.history.forEach(pos => this.tracerGfx.lineTo(pos.x, pos.y));
            this.tracerGfx.strokePath();
        }
        this.tracerGfx.fillStyle(config.headColor ?? 0x00ffff, alpha);
        this.tracerGfx.fillCircle(p.x, p.y, config.headSize ?? 2);
    }

    spawnParticle(config) {
        const life = 40 + Math.random() * 40;
        const push = config.push ?? 0.15;

        let sx = this.gameObject.x;
        let sy = this.gameObject.y;

        // If push is negative (Absorb), spawn in a ring around the center
        if (push < 0) {
            const angle = Math.random() * Math.PI * 2;
            const dist = 50 + Math.random() * 30;
            sx += Math.cos(angle) * dist;
            sy += Math.sin(angle) * dist;
        }

        this.particles.push({
            x: sx, y: sy,
            vx: (Math.random() - 0.5) * 2,
            vy: (Math.random() - 0.5) * 2,
            history: [], life: life, maxLife: life
        });
    }

    destroy() {
        if (this.tracerGfx) this.tracerGfx.destroy();
        this.particles = [];
        super.destroy();
    }
}

            angle: { min: 0, max: 360 },
            scale: { start: 0.4, end: 0 },

        const updateProp = (prop, min, max) => {
            if (prop && typeof prop === 'object') {
                if (prop.hasOwnProperty('min')) prop.min = min;
                if (prop.hasOwnProperty('max')) prop.max = max;
            }
        };

        const baseSpeed = 40 * intensity;
        updateProp(this.emitter.speedX, -baseSpeed, baseSpeed);
        updateProp(this.emitter.speedY, -baseSpeed, baseSpeed);

        this.emitter.frequency = Math.max(5, 60 / intensity);

        this.emitter.forEachAlive((p) => {
            const noise = p.id || 0;
            p.vx += Math.cos(time + noise) * (0.8 * intensity);
            p.vy += Math.sin(time + noise) * (0.8 * intensity);

            const dx = p.x - x;
            const dy = p.y - y;
            const dist = Math.sqrt(dx * dx + dy * dy);

            if (dist > 0) {
                const pushForce = (config.push ?? 0.3) * intensity;
                p.vx += (dx / dist) * pushForce;
                p.vy += (dy / dist) * pushForce;
            }
        });

        const baseRadius = (config.size || 1) * GRID.UNIT;
        this.gameObject.setRadius(baseRadius * (0.8 + intensity * 0.4));
    }

    setVisible(visible) {
        super.setVisible(visible);
        if (this.emitter) {
            this.emitter.setVisible(visible);
            this.emitter.paused = !visible;

import { BaseGraphic } from './BaseGraphic';
import { GRID } from '../core/Constants';
import Phaser from 'phaser';

export class TriangleGraphic extends BaseGraphic {
    create(config) {
        // Use a Graphics object instead of a Polygon Shape.
        // Graphics pivot exactly at (0,0) by default, ensuring a perfect "spin".
        return this.scene.add.graphics();
    }

    onUpdate(config) {
        const gfx = this.gameObject;
        if (!gfx) return;

        // 1. CLEAR previous frame (essential for dynamic zoom thickness)
        gfx.clear();

        // 2. CONFIGURATION
        const cam = this.scene.cameras.main;
        // Stroke stays 2px thick regardless of zoom
        const thickness = 2 / cam.zoom;
        const colorVal = Phaser.Display.Color.HexStringToColor(config.color || '#ff8400').color;
        const intensity = config.intensity ?? 1.0;

        // 3. SIZE CALCULATION
        // "10x10 World Units" means 10 * GRID.UNIT (10px) = 100px total span.
        // To span 100px centered at 0, radius 's' must be 50.
        // config.size of 5.0 * 10 = 50.
        const unit = GRID.UNIT;
        const s = (config.size || 1) * unit * intensity;

        // 4. DRAW
        // We draw relative to local (0,0), which BaseGraphic has already moved to the Player's position.
        gfx.fillStyle(colorVal, 1);
        gfx.lineStyle(thickness, 0xffffff);

        gfx.beginPath();
        gfx.moveTo(s, 0);       // Tip (Right)
        gfx.lineTo(-s, -s);     // Top Back
        gfx.lineTo(-s, s);      // Bottom Back
        gfx.closePath();

        gfx.fillPath();
        gfx.strokePath();
    }
}

export class MenuScene extends Phaser.Scene {
    constructor() {
        super('MenuScene');
        this.options = ['START GAME', 'LOAD GAME', 'TEST SUITE'];
        this.selected = 0;
        this.textObjects = [];
    }

    create() {
        this.inputManager = new InputManager(this);
        this.textObjects = []; // CRITICAL: Reset the array

        this.add.text(100, 100, 'REFRACT', {
            fontSize: '64px',
            fill: '#fff',
            fontFamily: 'monospace'
        });

        this.options.forEach((opt, index) => {
            const text = this.add.text(100, 300 + (index * 60), `  ${opt}`, {
                fontSize: '32px',
                fill: '#666',
                fontFamily: 'monospace'
            });
            this.textObjects.push(text);
        });

        this.updateSelection();
    }

    update() {
        if (this.inputManager.getAction('UP')) {
            this.selected = (this.selected - 1 + this.options.length) % this.options.length;
            this.updateSelection();
        }
        else if (this.inputManager.getAction('DOWN')) {
            this.selected = (this.selected + 1) % this.options.length;
            this.updateSelection();
        }

        if (this.inputManager.getAction('SELECT')) {
            this.handleSelection();
        }
    }

    updateSelection() {
        this.textObjects.forEach((text, index) => {
            // Safety check to prevent the drawImage error during scene transitions
            if (!text || !text.active || !text.canvas) return;

            if (index === this.selected) {
                text.setColor('#0f0');
                text.setText(`> ${this.options[index]}`);
            } else {
                text.setColor('#666');
                text.setText(`  ${this.options[index]}`);
            }
        });
    }

    handleSelection() {
        const choice = this.options[this.selected];
        if (choice === 'TEST SUITE') {
            this.scene.start('TestMenuScene');
        } else {
            console.log("Not implemented yet:", choice);
        }
    }
}

    }

    create() {

        this.inputManager = new InputManager(this);

        // Initialize UI Overlay
        UI.showTitle(this.testTitle, this.testSubtitle);

        // Global Key Listeners
        this.input.keyboard.on('keydown-TAB', () => UI.toggleDebug());
        this.input.keyboard.on('keydown-ESC', () => this.scene.start('TestMenuScene'));

        this.events.once('shutdown', this.shutdown, this);

        // Run sub-scene specific setup
        this.setupScene();

    }

        this.entities = [];
        this.input.keyboard.off('keydown-TAB');
        this.input.keyboard.off('keydown-ESC');
        this.accumulator = 0;
    }

    /**
     * The Heart of the ECS: The Game Loop
     */
    update(time, delta) {
        if (!this.sys.isActive()) return;

        this.accumulator += delta;

        // --- STEP 1: THE SIMULATION (Fixed Timestep) ---
        // We run these at 60Hz to ensure slingshot math is consistent.
        while (this.accumulator >= PHYSICS.STEP) {
            const dtSeconds = PHYSICS.STEP / 1000;

            // Intent Systems (Capture Stick/Buttons)
            SlingshotSystem(this.entities, this.inputManager);
            WebSystem(this.entities, this.inputManager);
            AbilityTargetingSystem(this.entities, this.inputManager);

            // Constraint Systems (Parenting & Dwell)
            AttachmentSystem(this.entities);

            // Motion Systems (Walking & Gravity)
            MovementSystem(this.entities, this.inputManager, dtSeconds);

            // Integration System (Calculates final X/Y based on Parents/Floor)
            PhysicsSystem(this.entities, PHYSICS.STEP);

            this.fixedUpdate(time, PHYSICS.STEP);
            this.accumulator -= PHYSICS.STEP;
        }

        // --- STEP 2: THE RENDER PASS ---
        // CRITICAL ORDER: Sync Phaser Sprites to ECS Position *before* drawing overlays.

        // 1. Sync Sprites/Triangles to their new ECS Positions
        LayeredRenderSystem(this.entities, this);

        // 2. Draw procedural graphics (Indicator/Webs/Targets) on top of the synced sprites
        SlingshotRenderSystem(this.entities, this);
        WebRenderSystem(this.entities, this);
        TargetRenderSystem(this.entities, this);

        // 3. Update Camera and HUD
        CameraSystem(this.entities, this);
        DebugLayerSystem(this.entities, this);

        // 4. Update the Telemetry UI
        this.updateDebugUI();
    }

    fixedUpdate(time, dt) { }

            const sling = player.components.SlingshotJump || {};

                }
            });
        }
    }

    addEntity(componentsArray) {

            components: {}
        };

            if (c && c._type) {
                entity.components[c._type] = c;
            }
        });

        this.entities.push(entity);
        return entity;
    }

            this.scene.start('MenuScene');

            if (!text || !text.active) return;

                text.setColor('#0f0').setText(`> ${sceneName}`);
            } else {
                text.setColor('#666').setText(`  ${sceneName}`);
            }
        });

// src/utils/CollisionUtils.js
import Phaser from 'phaser';

export const isOverlapping = (newPos, newCol, existingEntities) => {
    const platforms = existingEntities.filter(e => e.components.PlatformTag);
    const buffer = 20; // Extra space between platforms

    for (const plat of platforms) {
        const otherPos = plat.components.Position;
        const otherCol = plat.components.Collider;

        // 1. Circle to Circle
        if (newCol.type === 'circle' && otherCol.type === 'circle') {
            const dist = Phaser.Math.Distance.Between(newPos.x, newPos.y, otherPos.x, otherPos.y);
            if (dist < (newCol.radius + otherCol.radius + buffer)) return true;
        }
        // 2. AABB to AABB (Rectangles)
        else {
            const newW = newCol.width || newCol.radius * 2;
            const newH = newCol.height || newCol.radius * 2;
            const otherW = otherCol.width || otherCol.radius * 2;
            const otherH = otherCol.height || otherCol.radius * 2;

            const overlap = (
                Math.abs(newPos.x - otherPos.x) * 2 < (newW + otherW + buffer) &&
                Math.abs(newPos.y - otherPos.y) * 2 < (newH + otherH + buffer)
            );
            if (overlap) return true;
        }
    }
    return false;
};
